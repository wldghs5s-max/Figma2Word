2. ARCHITECTURE & REPOSITORY SUMMARY
================================================================
Figma2Word is a high-fidelity local-first document generation engine.
It converts Figma design ASTs into fully editable, pixel-accurate Microsoft Word (.docx)
and Hancom Hangul (.hwpx) documents without external paid APIs.

Core Components:
- Canvas IR (`src/canvas/types.ts`): 2D normalized coordinate intermediate representation.
- SVG Path Parser (`src/canvas/geometry/svgPathParser.ts`): Full W3C SVG parser converting
  complex paths (M, L, C, S, Q, T, A, Z) into normalized VML and HWPX polygon vertices.
- Figma Parser (`src/canvas/parser/figmaCanvasParser.ts`): AST -> Canvas IR with scale policies.
- DOCX Parser (`src/canvas/parser/docxCanvasParser.ts`): DOCX VML -> Canvas IR for direct DOCX->HWPX.
- DOCX Renderer (`src/canvas/renderer/docxCanvasRenderer.ts`): Canvas IR -> VML absolute shapes.
- HWPX Renderer (`src/canvas/renderer/hwpxCanvasRenderer.ts`): Canvas IR -> KS X 6101 OCF HWPX.
- Web UI (`apps/web/src/App.tsx`): React 19 + Vite 8.3 local web dashboard with dual engine
  toggles, DOCX/HWPX format selectors, and DOCX->HWPX one-click conversion.
- Quality: Real Word COM SSIM 0.8562 (peak 0.8768), 100% editability pass rate, 304/304 unit tests.

/**
 * Standard unit conversions:
 * Internal unit is PostScript point (pt, 1/72 inch).
 * 
 * - 1 pt = 20 twips (dxa)
 * - 1 pt = 12,700 EMUs (English Metric Units)
 * - 1 pt = 100 HWPUs (Hancom Word Processor Units)
 * - 1 pt = 25.4 / 72 mm ≈ 0.352778 mm
 * - 1 mm = 72 / 25.4 pt ≈ 2.834646 pt
 * - 1 CSS px (96 DPI) = 72 / 96 pt = 0.75 pt
 */

export function ptToTwip(pt: number): number {
  return Math.round(pt * 20);
}

export function twipToPt(twip: number): number {
  return twip / 20;
}

export function ptToEmu(pt: number): number {
  return Math.round(pt * 12700);
}

export function emuToPt(emu: number): number {
  return emu / 12700;
}

export function ptToHwpu(pt: number): number {
  return Math.round(pt * 100);
}

export function hwpuToPt(hwpu: number): number {
  return hwpu / 100;
}

  return (pt * 25.4) / 72;
}

  return (mm * 72) / 25.4;
}

}

export type ScalePolicy = "css-96dpi" | "fit-a4" | "1:1";

/**
 * Resolves scaling from Figma frame dimensions to output canvas dimensions.
 * Standard A4 portrait in pt: 595.28 x 841.89 pt (210 x 297 mm).
 */
export function resolveScale(
  frameWidthPx: number,
  frameHeightPx: number,
  policy: ScalePolicy = "fit-a4"
): ResolvedScale {
  const A4_WIDTH_PT = 595.28;
  const A4_HEIGHT_PT = 841.89;

  if (policy === "1:1") {
    return {
      scaleX: 1.0,
      scaleY: 1.0,
      targetWidthPt: frameWidthPx,
      targetHeightPt: frameHeightPx,
      policy,
    };
  }

  if (policy === "css-96dpi") {
    const scale = 0.75; // 72 / 96
    return {
      scaleX: scale,
      scaleY: scale,
      targetWidthPt: frameWidthPx * scale,
      targetHeightPt: frameHeightPx * scale,
      policy,
    };
  }

  // policy === 'fit-a4'
  // If frame aspect ratio is close to A4 (0.68 ~ 0.73), scale to fit exact A4 width/height
  const ratio = frameWidthPx / frameHeightPx;
  const isA4Portrait = ratio >= 0.65 && ratio <= 0.75;
  const isA4Landscape = ratio >= 1.35 && ratio <= 1.50;

  if (isA4Portrait) {
    const scaleX = A4_WIDTH_PT / frameWidthPx;
    const scaleY = A4_HEIGHT_PT / frameHeightPx;
    return {
      scaleX,
      scaleY,
      targetWidthPt: A4_WIDTH_PT,
      targetHeightPt: A4_HEIGHT_PT,
      policy,
    };
  }

  if (isA4Landscape) {
    const scaleX = A4_HEIGHT_PT / frameWidthPx;
    const scaleY = A4_WIDTH_PT / frameHeightPx;
    return {
      scaleX,
      scaleY,
      targetWidthPt: A4_HEIGHT_PT,
      targetHeightPt: A4_WIDTH_PT,
      policy,
    };
  }

  // Fallback to standard 96 DPI CSS conversion
  const scale = 0.75;
  return {
    scaleX: scale,
    scaleY: scale,
    targetWidthPt: frameWidthPx * scale,
    targetHeightPt: frameHeightPx * scale,
    policy,
  };
}

/**
 * SVG Path Parser & Geometry Transpiler for Canvas IR, VML, and OWPML (HWPX).
 *
 * Supports SVG path commands:
 * - M / m (Move to)
 * - L / l (Line to)
 * - H / h (Horizontal line to)
 * - V / v (Vertical line to)
 * - C / c (Cubic Bezier)
 * - S / s (Smooth cubic Bezier)
 * - Q / q (Quadratic Bezier)
 * - T / t (Smooth quadratic Bezier)
 * - A / a (Elliptical Arc)
 * - Z / z (Close path)
 */

export interface Point2D {
  x: number;
  y: number;
}

export type NormalizedCommandType = "M" | "L" | "C" | "Z";

export interface NormalizedPathCommand {
  type: NormalizedCommandType;
  // For 'M' and 'L': [p1]
  // For 'C': [cp1, cp2, endPoint]
  // For 'Z': []
  points: Point2D[];
}

export interface PathBounds {
  minX: number;
  minY: number;
  maxX: number;
  maxY: number;
  width: number;
  height: number;
}

/**
 * Tokenize an SVG path string into command letters and numerical tokens.
 */
function tokenizeSvgPath(d: string): (string | number)[] {
  const tokens: (string | number)[] = [];
  // Matches command letters, or floats including scientific notation and negative numbers
  const regex = /([a-df-z])|([-+]?(?:\d*\.\d+|\d+)(?:[eE][-+]?\d+)?)/gi;
  let match: RegExpExecArray | null;

  while ((match = regex.exec(d)) !== null) {
    if (match[1] !== undefined) {
      tokens.push(match[1]);
    } else if (match[2] !== undefined) {
      const num = parseFloat(match[2]);
      if (!isNaN(num)) {
        tokens.push(num);
      }
    }
  }
  return tokens;
}

/**
 * Convert an SVG elliptical arc segment into one or more cubic Bezier curve commands.
 * Based on the W3C SVG 1.1 implementation notes (F.6 Elliptical arc implementation notes).
 */
function arcToCubicBeziers(
  x1: number,
  y1: number,
  rx: number,
  ry: number,
  angleDegrees: number,
  largeArcFlag: boolean,
  sweepFlag: boolean,
  x2: number,
  y2: number
): NormalizedPathCommand[] {
  const commands: NormalizedPathCommand[] = [];

  // If endpoints are identical, arc is degenerate
  if (Math.abs(x1 - x2) < 1e-6 && Math.abs(y1 - y2) < 1e-6) {
    return commands;
  }

  // Ensure radii are positive
  rx = Math.abs(rx);
  ry = Math.abs(ry);
  if (rx < 1e-6 || ry < 1e-6) {
    // Treat as straight line
    commands.push({ type: "L", points: [{ x: x2, y: y2 }] });
    return commands;
  }

  const phi = (angleDegrees * Math.PI) / 180;
  const cosPhi = Math.cos(phi);
  const sinPhi = Math.sin(phi);

  // Step 1: Compute (x1', y1')
  const dx = (x1 - x2) / 2;
  const dy = (y1 - y2) / 2;
  const x1p = cosPhi * dx + sinPhi * dy;
  const y1p = -sinPhi * dx + cosPhi * dy;

  // Correct radii if necessary
  const lambda = (x1p * x1p) / (rx * rx) + (y1p * y1p) / (ry * ry);
  if (lambda > 1) {
    const sqrtLambda = Math.sqrt(lambda);
    rx *= sqrtLambda;
    ry *= sqrtLambda;
  }

  // Step 2: Compute (cx', cy')
  const sign = largeArcFlag === sweepFlag ? -1 : 1;
  const sq =
    Math.max(
      0,
      rx * rx * (ry * ry) - rx * rx * (y1p * y1p) - ry * ry * (x1p * x1p)
    ) / (rx * rx * (y1p * y1p) + ry * ry * (x1p * x1p));
  const coef = sign * Math.sqrt(sq);
  const cxp = coef * ((rx * y1p) / ry);
  const cyp = coef * (-(ry * x1p) / rx);

  // Step 3: Compute (cx, cy) from (cx', cy')
  const cx = cosPhi * cxp - sinPhi * cyp + (x1 + x2) / 2;
  const cy = sinPhi * cxp + cosPhi * cyp + (y1 + y2) / 2;

  // Step 4: Compute theta1 and deltaTheta
  const v1x = (x1p - cxp) / rx;
  const v1y = (y1p - cyp) / ry;
  const v2x = (-x1p - cxp) / rx;
  const v2y = (-y1p - cyp) / ry;

  function vectorAngle(ux: number, uy: number, vx: number, vy: number): number {
    const dot = ux * vx + uy * vy;
    const len = Math.sqrt(ux * ux + uy * uy) * Math.sqrt(vx * vx + vy * vy);
    let ang = Math.acos(Math.max(-1, Math.min(1, dot / (len || 1))));
    if (ux * vy - uy * vx < 0) ang = -ang;
    return ang;
  }

  const theta1 = vectorAngle(1, 0, v1x, v1y);
  let deltaTheta = vectorAngle(v1x, v1y, v2x, v2y);

  if (!sweepFlag && deltaTheta > 0) {
    deltaTheta -= 2 * Math.PI;
  } else if (sweepFlag && deltaTheta < 0) {
    deltaTheta += 2 * Math.PI;
  }

  // Split arc into segments of at most pi/2
  const segments = Math.max(1, Math.ceil(Math.abs(deltaTheta) / (Math.PI / 2)));
  const dTheta = deltaTheta / segments;
  const halfDTheta = dTheta / 2;
  const alpha =
    (Math.sin(dTheta) * (Math.sqrt(4 + 3 * Math.tan(halfDTheta) ** 2) - 1)) / 3;

  let currentTheta = theta1;

  for (let i = 0; i < segments; i++) {
    const nextTheta = currentTheta + dTheta;

    const cosTheta = Math.cos(currentTheta);
    const sinTheta = Math.sin(currentTheta);
    const cosNextTheta = Math.cos(nextTheta);
    const sinNextTheta = Math.sin(nextTheta);

    const ep1x = cosPhi * rx * cosTheta - sinPhi * ry * sinTheta + cx;
    const ep1y = sinPhi * rx * cosTheta + cosPhi * ry * sinTheta + cy;

    const ep2x = cosPhi * rx * cosNextTheta - sinPhi * ry * sinNextTheta + cx;
    const ep2y = sinPhi * rx * cosNextTheta + cosPhi * ry * sinNextTheta + cy;

    const dTheta1x = -cosPhi * rx * sinTheta - sinPhi * ry * cosTheta;
    const dTheta1y = -sinPhi * rx * sinTheta + cosPhi * ry * cosTheta;

    const dTheta2x = -cosPhi * rx * sinNextTheta - sinPhi * ry * cosNextTheta;
    const dTheta2y = -sinPhi * rx * sinNextTheta + cosPhi * ry * cosNextTheta;

    const cp1 = { x: ep1x + alpha * dTheta1x, y: ep1y + alpha * dTheta1y };
    const cp2 = { x: ep2x - alpha * dTheta2x, y: ep2y - alpha * dTheta2y };
    const targetPt = i === segments - 1 ? { x: x2, y: y2 } : { x: ep2x, y: ep2y };

    commands.push({
      type: "C",
      points: [cp1, cp2, targetPt],
    });

    currentTheta = nextTheta;
  }

  return commands;
}

export class SvgPath {
  public commands: NormalizedPathCommand[];

  constructor(commands: NormalizedPathCommand[] = []) {
    this.commands = commands;
  }

  /**
   * Parse any SVG path string into normalized absolute commands (M, L, C, Z).
   */
  public static parse(d: string): SvgPath {
    if (!d || typeof d !== "string") {
      return new SvgPath([]);
    }

    const tokens = tokenizeSvgPath(d);
    const normalized: NormalizedPathCommand[] = [];

    let curX = 0;
    let curY = 0;
    let startX = 0;
    let startY = 0;
    let lastCpX = 0;
    let lastCpY = 0;
    let lastCmd = "";

    let idx = 0;
    while (idx < tokens.length) {
      const tok = tokens[idx];
      if (typeof tok !== "string") {
        idx++;
        continue;
      }

      let cmd = tok;
      idx++;

      while (idx < tokens.length && typeof tokens[idx] === "number") {
        const isRelative = cmd >= "a" && cmd <= "z";
        const upper = cmd.toUpperCase();

        switch (upper) {
          case "M": {
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);
            curX = x;
            curY = y;
            startX = x;
            startY = y;
            lastCpX = x;
            lastCpY = y;
            normalized.push({ type: "M", points: [{ x, y }] });
            // Subsequent coordinate pairs treated as implicit L / l
            cmd = isRelative ? "l" : "L";
            break;
          }
          case "L": {
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);
            curX = x;
            curY = y;
            lastCpX = x;
            lastCpY = y;
            normalized.push({ type: "L", points: [{ x, y }] });
            break;
          }
          case "H": {
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            curX = x;
            lastCpX = x;
            lastCpY = curY;
            normalized.push({ type: "L", points: [{ x, y: curY }] });
            break;
          }
          case "V": {
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);
            curY = y;
            lastCpX = curX;
            lastCpY = y;
            normalized.push({ type: "L", points: [{ x: curX, y }] });
            break;
          }
          case "C": {
            const x1 = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y1 = (tokens[idx++] as number) + (isRelative ? curY : 0);
            const x2 = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y2 = (tokens[idx++] as number) + (isRelative ? curY : 0);
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);

            normalized.push({
              type: "C",
              points: [
                { x: x1, y: y1 },
                { x: x2, y: y2 },
                { x, y },
              ],
            });
            lastCpX = x2;
            lastCpY = y2;
            curX = x;
            curY = y;
            break;
          }
          case "S": {
            // Smooth cubic: cp1 is reflection of lastCp about cur
            let x1 = curX;
            let y1 = curY;
            if (lastCmd === "C" || lastCmd === "S") {
              x1 = 2 * curX - lastCpX;
              y1 = 2 * curY - lastCpY;
            }
            const x2 = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y2 = (tokens[idx++] as number) + (isRelative ? curY : 0);
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);

            normalized.push({
              type: "C",
              points: [
                { x: x1, y: y1 },
                { x: x2, y: y2 },
                { x, y },
              ],
            });
            lastCpX = x2;
            lastCpY = y2;
            curX = x;
            curY = y;
            break;
          }
          case "Q": {
            // Quadratic Bezier -> convert to cubic
            const qx = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const qy = (tokens[idx++] as number) + (isRelative ? curY : 0);
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);

            const x1 = curX + (2 / 3) * (qx - curX);
            const y1 = curY + (2 / 3) * (qy - curY);
            const x2 = x + (2 / 3) * (qx - x);
            const y2 = y + (2 / 3) * (qy - y);

            normalized.push({
              type: "C",
              points: [
                { x: x1, y: y1 },
                { x: x2, y: y2 },
                { x, y },
              ],
            });
            lastCpX = qx;
            lastCpY = qy;
            curX = x;
            curY = y;
            break;
          }
          case "T": {
            // Smooth quadratic Bezier -> convert to cubic
            let qx = curX;
            let qy = curY;
            if (lastCmd === "Q" || lastCmd === "T") {
              qx = 2 * curX - lastCpX;
              qy = 2 * curY - lastCpY;
            }
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);

            const x1 = curX + (2 / 3) * (qx - curX);
            const y1 = curY + (2 / 3) * (qy - curY);
            const x2 = x + (2 / 3) * (qx - x);
            const y2 = y + (2 / 3) * (qy - y);

            normalized.push({
              type: "C",
              points: [
                { x: x1, y: y1 },
                { x: x2, y: y2 },
                { x, y },
              ],
            });
            lastCpX = qx;
            lastCpY = qy;
            curX = x;
            curY = y;
            break;
          }
          case "A": {
            const rx = tokens[idx++] as number;
            const ry = tokens[idx++] as number;
            const rot = tokens[idx++] as number;
            const largeArc = (tokens[idx++] as number) !== 0;
            const sweep = (tokens[idx++] as number) !== 0;
            const x = (tokens[idx++] as number) + (isRelative ? curX : 0);
            const y = (tokens[idx++] as number) + (isRelative ? curY : 0);

            const arcBeziers = arcToCubicBeziers(
              curX,
              curY,
              rx,
              ry,
              rot,
              largeArc,
              sweep,
              x,
              y
            );
            for (const b of arcBeziers) {
              normalized.push(b);
            }
            curX = x;
            curY = y;
            lastCpX = x;
            lastCpY = y;
            break;
          }
          case "Z": {
            normalized.push({ type: "Z", points: [] });
            curX = startX;
            curY = startY;
            lastCpX = curX;
            lastCpY = curY;
            break;
          }
          default: {
            idx++;
            break;
          }
        }
        lastCmd = upper;
      }

      if (cmd.toUpperCase() === "Z") {
        normalized.push({ type: "Z", points: [] });
        curX = startX;
        curY = startY;
        lastCpX = curX;
        lastCpY = curY;
        lastCmd = "Z";
      }
    }

    return new SvgPath(normalized);
  }

  /**
   * Compute exact bounding box across all points.
   */
  public getBounds(): PathBounds {
    if (this.commands.length === 0) {
      return { minX: 0, minY: 0, maxX: 0, maxY: 0, width: 0, height: 0 };
    }

    let minX = Infinity;
    let minY = Infinity;
    let maxX = -Infinity;
    let maxY = -Infinity;

    for (const cmd of this.commands) {
      for (const pt of cmd.points) {
        if (pt.x < minX) minX = pt.x;
        if (pt.x > maxX) maxX = pt.x;
        if (pt.y < minY) minY = pt.y;
        if (pt.y > maxY) maxY = pt.y;
      }
    }

    if (minX === Infinity) {
      return { minX: 0, minY: 0, maxX: 0, maxY: 0, width: 0, height: 0 };
    }

    return {
      minX,
      minY,
      maxX,
      maxY,
      width: Math.max(1e-6, maxX - minX),
      height: Math.max(1e-6, maxY - minY),
    };
  }

  /**
   * Convert normalized SVG path into a standard VML path string scaled to coordsize.
   * e.g. "m 0,0 l 1000,0 l 1000,1000 x e"
   */
  public toVmlPath(targetCoordWidth = 1000, targetCoordHeight = 1000): string {
    if (this.commands.length === 0) {
      return "";
    }

    const bounds = this.getBounds();
    const scaleX = targetCoordWidth / bounds.width;
    const scaleY = targetCoordHeight / bounds.height;

    const parts: string[] = [];

    for (const cmd of this.commands) {
      switch (cmd.type) {
        case "M": {
          const pt = cmd.points[0];
          const vx = Math.round((pt.x - bounds.minX) * scaleX);
          const vy = Math.round((pt.y - bounds.minY) * scaleY);
          parts.push(`m ${vx},${vy}`);
          break;
        }
        case "L": {
          const pt = cmd.points[0];
          const vx = Math.round((pt.x - bounds.minX) * scaleX);
          const vy = Math.round((pt.y - bounds.minY) * scaleY);
          parts.push(`l ${vx},${vy}`);
          break;
        }
        case "C": {
          const [cp1, cp2, ep] = cmd.points;
          const cx1 = Math.round((cp1.x - bounds.minX) * scaleX);
          const cy1 = Math.round((cp1.y - bounds.minY) * scaleY);
          const cx2 = Math.round((cp2.x - bounds.minX) * scaleX);
          const cy2 = Math.round((cp2.y - bounds.minY) * scaleY);
          const ex = Math.round((ep.x - bounds.minX) * scaleX);
          const ey = Math.round((ep.y - bounds.minY) * scaleY);
          parts.push(`c ${cx1},${cy1},${cx2},${cy2},${ex},${ey}`);
          break;
        }
        case "Z": {
          parts.push("x e");
          break;
        }
      }
    }

    if (!parts[parts.length - 1]?.endsWith("e")) {
      parts.push("e");
    }

    return parts.join(" ");
  }

  /**
   * Sample path into discrete polygon vertices for HWPX (<hp:pt x="..." y="..."/>).
   * Cubic curves are sampled into intermediate vertices based on steps.
   */
  public toHwpxPolygonPoints(
    targetWidthHwpu: number,
    targetHeightHwpu: number,
    stepsPerCurve = 6
  ): Point2D[] {
    if (this.commands.length === 0) {
      return [];
    }

    const bounds = this.getBounds();
    const scaleX = targetWidthHwpu / bounds.width;
    const scaleY = targetHeightHwpu / bounds.height;

    const pts: Point2D[] = [];
    let curX = bounds.minX;
    let curY = bounds.minY;

    for (const cmd of this.commands) {
      switch (cmd.type) {
        case "M":
        case "L": {
          const p = cmd.points[0];
          curX = p.x;
          curY = p.y;
          pts.push({
            x: Math.round((p.x - bounds.minX) * scaleX),
            y: Math.round((p.y - bounds.minY) * scaleY),
          });
          break;
        }
        case "C": {
          const [cp1, cp2, ep] = cmd.points;
          // Sample cubic Bezier B(t) = (1-t)^3 P0 + 3(1-t)^2 t P1 + 3(1-t) t^2 P2 + t^3 P3
          for (let step = 1; step <= stepsPerCurve; step++) {
            const t = step / stepsPerCurve;
            const omt = 1 - t;
            const b0 = omt * omt * omt;
            const b1 = 3 * omt * omt * t;
            const b2 = 3 * omt * t * t;
            const b3 = t * t * t;

            const px = b0 * curX + b1 * cp1.x + b2 * cp2.x + b3 * ep.x;
            const py = b0 * curY + b1 * cp1.y + b2 * cp2.y + b3 * ep.y;

            pts.push({
              x: Math.round((px - bounds.minX) * scaleX),
              y: Math.round((py - bounds.minY) * scaleY),
            });
          }
          curX = ep.x;
          curY = ep.y;
          break;
        }
        case "Z": {
          // Polygon is closed automatically in HWPX
          break;
        }
      }
    }

    return pts;
  }

import {
  CanvasDocument,
  CanvasPage,
  CanvasElement,
  CanvasTextRun,
  CanvasStyle,
  ScalePolicy,
  resolveScale,
  CanvasResource,
} from "../types.js";
import { FigmaNode, FigmaFileResponse, FigmaColor, FigmaPaint } from "../../parser/types.js";
import { isA4Ratio } from "../../pipeline/index.js";
import { SvgPath } from "../geometry/svgPathParser.js";

export interface FigmaCanvasParserOptions {
  imageMap?: Record<string, string>;
  selectedFrameIds?: string[];
  scalePolicy?: ScalePolicy;
  titleOverride?: string;
}

export class FigmaCanvasParser {
  private imageMap: Record<string, string>;
  private scalePolicy: ScalePolicy;
  private selectedFrameIds?: Set<string>;
  private titleOverride?: string;
  private resources: Record<string, CanvasResource> = {};
  private imageCounter = 0;
  private imageRefMap = new Map<string, string>(); // figmaRef -> resourceId

  constructor(options?: FigmaCanvasParserOptions) {
    this.imageMap = options?.imageMap || {};
    this.scalePolicy = options?.scalePolicy || "fit-a4";
    this.selectedFrameIds = options?.selectedFrameIds
      ? new Set(options.selectedFrameIds)
      : undefined;
    this.titleOverride = options?.titleOverride;
  }

  public parse(input: FigmaNode | FigmaFileResponse): CanvasDocument {
    this.resources = {};
    this.imageCounter = 0;
    this.imageRefMap.clear();

    const rootNode = "document" in input ? input.document : input;
    const documentName =
      this.titleOverride ||
      ("name" in input && input.name ? input.name : "Figma Document");

    const notices: string[] = [];
    const warnings: string[] = [];

    // Step 1: Discover top-level frames representing document pages
    const pageFrames = this.discoverPageFrames(rootNode);

    if (pageFrames.length === 0) {
      warnings.push("No candidate page frames found. Using root as single page.");
    }

    const pages: CanvasPage[] = [];

    // If no page frames found, treat root itself as a page
    const framesToProcess = pageFrames.length > 0 ? pageFrames : [rootNode];

    for (let i = 0; i < framesToProcess.length; i++) {
      const frame = framesToProcess[i];
      const pageIndex = i + 1;
      const page = this.parsePageFrame(frame, pageIndex, notices, warnings);
      pages.push(page);
    }

    return {
      title: documentName,
      pageSize: {
        name: "A4",
        widthPt: 595.28,
        heightPt: 841.89,
        unit: "pt",
      },
      marginsPt: {
        top: 0,
        right: 0,
        bottom: 0,
        left: 0,
      },
      pages,
      resources: this.resources,
      diagnostics: {
        notices,
        warnings,
      },
    };
  }

  /**
   * Discovers top-level frames that qualify as document pages (e.g. A4 frames).
   */
  private discoverPageFrames(root: FigmaNode): FigmaNode[] {
    const candidateFrames: FigmaNode[] = [];

    const visitCanvasOrDoc = (node: FigmaNode) => {
      if (node.children) {
        for (const child of node.children) {
          if (child.visible === false) continue;

          // If frame selection filter is active
          if (this.selectedFrameIds && !this.selectedFrameIds.has(child.id)) {
            continue;
          }

          // Check if child is a frame or component
          if (
            child.type === "FRAME" ||
            child.type === "COMPONENT" ||
            child.type === "GROUP"
          ) {
            candidateFrames.push(child);
          } else if (child.type === "CANVAS") {
            visitCanvasOrDoc(child);
          }
        }
      }
    };

    if (root.type === "DOCUMENT" || root.type === "CANVAS") {
      visitCanvasOrDoc(root);
    } else {
      candidateFrames.push(root);
    }

    // Sort candidate frames in reading order (top-to-bottom, left-to-right)
    candidateFrames.sort((a, b) => {
      const boxA = a.absoluteBoundingBox || { x: 0, y: 0 };
      const boxB = b.absoluteBoundingBox || { x: 0, y: 0 };
      const rowA = Math.round(boxA.y / 100);
      const rowB = Math.round(boxB.y / 100);
      if (rowA !== rowB) return rowA - rowB;
      return boxA.x - boxB.x;
    });

    return candidateFrames;
  }

  /**
   * Parses a single top-level frame into a CanvasPage.
   */
  private parsePageFrame(
    frame: FigmaNode,
    pageNumber: number,
    notices: string[],
    warnings: string[]
  ): CanvasPage {
    const frameBox = frame.absoluteBoundingBox || {
      x: 0,
      y: 0,
      width: 794,
      height: 1123,
    };

    const resolved = resolveScale(
      frameBox.width,
      frameBox.height,
      this.scalePolicy
    );
    const { scaleX, scaleY, targetWidthPt, targetHeightPt } = resolved;

    // Check background color
    let backgroundColor: string | undefined;
    const rawBg = (frame as any).backgroundColor;
    if (rawBg) {
      backgroundColor = this.figmaColorToHex(rawBg);
    } else if (frame.fills && frame.fills.length > 0) {
      const solid = frame.fills.find(
        (f) => f.type === "SOLID" && f.visible !== false
      );
      if (solid && solid.color) {
        backgroundColor = this.figmaColorToHex(solid.color);
      }
    }

    const elements: CanvasElement[] = [];
    let currentZIndex = 1;

    // Helper to recursively process children
    const processChildren = (node: FigmaNode) => {
      if (!node.children) return;

      for (const child of node.children) {
        if (child.visible === false) continue;

        const childBox = child.absoluteBoundingBox;
        if (!childBox) {
          // If no absolute bounding box, still process children if any
          processChildren(child);
          continue;
        }

        // Relative coordinate to page frame
        const relX = childBox.x - frameBox.x;
        const relY = childBox.y - frameBox.y;

        const xPt = relX * scaleX;
        const yPt = relY * scaleY;
        const wPt = childBox.width * scaleX;
        const hPt = childBox.height * scaleY;

        // Diagnostic: clipsContent check
        if (child.clipsContent) {
          notices.push(
            `Page ${pageNumber}, node ${child.name || child.id} has clipsContent=true.`
          );
        }

        // Check if node is an image fill
        const imageFill = child.fills?.find(
          (f) => f.type === "IMAGE" && f.visible !== false && f.imageRef
        );

        if (imageFill && imageFill.imageRef) {
          const resId = this.resolveImageResource(imageFill.imageRef);
          elements.push({
            id: child.id,
            name: child.name,
            type: "image",
            bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
            zIndex: currentZIndex++,
            rotation: (child as any).rotation,
            opacity: child.opacity,
            style: this.extractStyle(child, scaleX),
            imageRef: resId,
            sourceNodeId: child.id,
            sourceNodeType: child.type,
          });
        } else if (child.type === "TEXT") {
          const textRuns = this.extractTextRuns(child, scaleX);
          const style = this.extractStyle(child, scaleX);
          const align = this.extractTextAlign(child);

          // For Figma TEXT nodes, 'fills' represents character font colors (stored in textRuns),
          // NOT a container background fill. Remove fillColor so the text box has a transparent background.
          delete style.fillColor;
          delete style.fillOpacity;

          elements.push({
            id: child.id,
            name: child.name,
            type: "text",
            bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
            zIndex: currentZIndex++,
            rotation: (child as any).rotation,
            opacity: child.opacity,
            style,
            textRuns,
            paragraphAlign: align,
            sourceNodeId: child.id,
            sourceNodeType: child.type,
          });
        } else if (child.type === "LINE") {
          elements.push({
            id: child.id,
            name: child.name,
            type: "line",
            bounds: { x: xPt, y: yPt, width: wPt, height: Math.max(hPt, 1) },
            zIndex: currentZIndex++,
            rotation: (child as any).rotation,
            opacity: child.opacity,
            style: this.extractStyle(child, scaleX),
            sourceNodeId: child.id,
            sourceNodeType: child.type,
          });
        } else if (child.type === "ELLIPSE") {
          const style = this.extractStyle(child, scaleX);
          const hasFill = !!style.fillColor;
          const hasStroke = !!style.strokeColor && (style.strokeWidth || 0) > 0;
          if (hasFill || hasStroke) {
            elements.push({
              id: child.id,
              name: child.name,
              type: "oval",
              bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
              zIndex: currentZIndex++,
              rotation: (child as any).rotation,
              opacity: child.opacity,
              style,
              sourceNodeId: child.id,
              sourceNodeType: child.type,
            });
          }
          processChildren(child);
        } else if (child.type === "VECTOR") {
          const style = this.extractStyle(child, scaleX);
          const hasFill = !!style.fillColor;
          const hasStroke = !!style.strokeColor && (style.strokeWidth || 0) > 0;

          if (hasFill || hasStroke) {
            const parentName = node.name || "";
            const nodeName = child.name || "";
            let rawPath = "";

            if ((child as any).fillGeometry?.[0]?.path) {
              rawPath = (child as any).fillGeometry[0].path;
            } else if ((child as any).vectorPaths?.[0]?.data) {
              rawPath = (child as any).vectorPaths[0].data;
            } else {
              // Standard UI icon path normalization (SVG format)
              const nameLower = (parentName + " " + nodeName).toLowerCase();
              if (nameLower.includes("alert") || nameLower.includes("warn") || nameLower.includes("triangle")) {
                rawPath = "M 500 60 L 950 920 L 50 920 Z M 500 380 L 500 640 M 500 780 L 500 840";
              } else if (nameLower.includes("check") || nameLower.includes("tick")) {
                rawPath = "M 150 520 L 400 780 L 880 180";
              } else if (nameLower.includes("info")) {
                rawPath = "M 500 50 C 748 50 950 252 950 500 C 950 748 748 950 500 950 C 252 950 50 748 50 500 C 50 252 252 50 500 50 Z M 500 280 L 500 340 M 500 460 L 500 740";
              } else if (nameLower.includes("file") || nameLower.includes("doc")) {
                rawPath = "M 200 80 L 620 80 L 820 280 L 820 920 L 200 920 Z M 620 80 L 620 280 L 820 280 M 340 480 L 680 480 M 340 640 L 680 640";
              }
            }

            let vmlPath = "";
            let vectorPoints: { x: number; y: number }[] | undefined;

            if (rawPath) {
              const parsed = SvgPath.parse(rawPath);
              vmlPath = parsed.toVmlPath(1000, 1000);
              const wHwpu = Math.max(10, Math.round(wPt * 100));
              const hHwpu = Math.max(10, Math.round(hPt * 100));
              vectorPoints = parsed.toHwpxPolygonPoints(wHwpu, hHwpu);
            }

            elements.push({
              id: child.id,
              name: child.name,
              type: "vector",
              bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
              zIndex: currentZIndex++,
              rotation: (child as any).rotation,
              opacity: child.opacity,
              style,
              vectorPath: vmlPath,
              rawSvgPath: rawPath,
              vectorPoints,
              sourceNodeId: child.id,
              sourceNodeType: child.type,
            });
          }
          processChildren(child);
        } else if (
          child.type === "RECTANGLE" ||
          child.type === "FRAME" ||
          child.type === "COMPONENT" ||
          child.type === "INSTANCE"
        ) {
          const style = this.extractStyle(child, scaleX);
          const hasFill = !!style.fillColor;
          const hasStroke = !!style.strokeColor && (style.strokeWidth || 0) > 0;

          // If frame has visible background/border, register card
          if (hasFill || hasStroke) {
            elements.push({
              id: child.id,
              name: child.name,
              type: "card",
              bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
              zIndex: currentZIndex++,
              rotation: (child as any).rotation,
              opacity: child.opacity,
              style,
              sourceNodeId: child.id,
              sourceNodeType: child.type,
            });
          }

          // Process nested children inside this container
          processChildren(child);
        } else if (child.type === "GROUP") {
          processChildren(child);
        } else {
          // Fallback for VECTOR, STAR, ELLIPSE, etc.
          const style = this.extractStyle(child, scaleX);
          if (style.fillColor || style.strokeColor) {
            elements.push({
              id: child.id,
              name: child.name,
              type: "card",
              bounds: { x: xPt, y: yPt, width: wPt, height: hPt },
              zIndex: currentZIndex++,
              rotation: (child as any).rotation,
              opacity: child.opacity,
              style,
              sourceNodeId: child.id,
              sourceNodeType: child.type,
            });
          }
          processChildren(child);
        }
      }
    };

    processChildren(frame);

    // Sort elements by zIndex
    elements.sort((a, b) => a.zIndex - b.zIndex);

    return {
      pageNumber,
      name: frame.name || `Page ${pageNumber}`,
      bounds: {
        x: 0,
        y: 0,
        width: targetWidthPt,
        height: targetHeightPt,
      },
      backgroundColor,
      elements,
    };
  }

  /**
   * Resolves a Figma imageRef to a local CanvasResource.
   */
  private resolveImageResource(imageRef: string): string {
    if (this.imageRefMap.has(imageRef)) {
      return this.imageRefMap.get(imageRef)!;
    }

    this.imageCounter++;
    const resId = `image${this.imageCounter}.png`;
    const dataUrl = this.imageMap[imageRef];

    let buffer: Buffer;
    let mimeType = "image/png";

    if (dataUrl && dataUrl.startsWith("data:")) {
      const match = dataUrl.match(/^data:([^;]+);base64,(.+)$/);
      if (match) {
        mimeType = match[1];
        buffer = Buffer.from(match[2], "base64");
      } else {
        buffer = Buffer.alloc(0);
      }
    } else {
      // 1x1 transparent placeholder
      buffer = Buffer.from(
        "iVBORw0KGgoAAAANSUhEUgAAAAEAAAABCAQAAAC1HAwCAAAAC0lEQVR42mNkYAAAAAYAAjCB0C8AAAAASUVORK5CYII=",
        "base64"
      );
    }

    this.resources[resId] = {
      id: resId,
      name: resId,
      mimeType,
      data: buffer,
    };

    this.imageRefMap.set(imageRef, resId);
    return resId;
  }

  /**
   * Extracts visual style (fill, stroke, corner radius) from a Figma node.
   */
  private extractStyle(node: FigmaNode, scale: number): CanvasStyle {
    const style: CanvasStyle = {};

    // 1. Fills
    if (node.fills && node.fills.length > 0) {
      const solid = node.fills.find(
        (f) => f.type === "SOLID" && f.visible !== false
      );
      if (solid && solid.color) {
        style.fillColor = this.figmaColorToHex(solid.color);
        style.fillOpacity = solid.opacity ?? solid.color.a ?? 1.0;
      }
    }

    // 2. Strokes
    if (node.strokes && node.strokes.length > 0) {
      const stroke = node.strokes.find(
        (s) => s.type === "SOLID" && s.visible !== false
      );
      if (stroke && stroke.color) {
        style.strokeColor = this.figmaColorToHex(stroke.color);
        style.strokeWidth = (node.strokeWeight || 1) * scale;
        const dashes = (node as any).strokeDashes;
        if (dashes && dashes.length > 0) {
          style.strokeDashed = true;
        }
      }
    }

    // 3. Corner Radius
    if (node.cornerRadius) {
      style.cornerRadius = node.cornerRadius * scale;
    }

    return style;
  }

  /**
   * Extracts text runs from a Figma TEXT node, taking into account
   * character style overrides if present.
   */
  private extractTextRuns(node: FigmaNode, scale: number): CanvasTextRun[] {
    const characters = node.characters || "";
    if (!characters) return [];

    const defaultFontFamily = node.style?.fontFamily || "Inter";
    const defaultFontSize = (node.style?.fontSize || 12) * scale;
    const rawWeight = node.style?.fontWeight;
    const defaultFontWeight =
      typeof rawWeight === "number"
        ? rawWeight
        : rawWeight
        ? parseInt(String(rawWeight), 10) || 400
        : 400;
    const defaultItalic =
      node.style?.italic ||
      node.style?.fontPostScriptName?.toLowerCase().includes("italic") ||
      false;
    const defaultUnderline = node.style?.textDecoration === "UNDERLINE";
    const defaultStrike = node.style?.textDecoration === "STRIKETHROUGH";

    // Text color from fills
    let defaultColor = "#000000";
    if (node.fills && node.fills.length > 0) {
      const solid = node.fills.find(
        (f) => f.type === "SOLID" && f.visible !== false
      );
      if (solid && solid.color) {
        defaultColor = this.figmaColorToHex(solid.color);
      }
    }

    const runs: CanvasTextRun[] = [];

    // Check for characterStyleOverrides
    const overrides = node.characterStyleOverrides;
    const styleTable = node.styleOverrideTable;

    if (
      overrides &&
      Array.isArray(overrides) &&
      overrides.length > 0 &&
      styleTable
    ) {
      let currentRunText = "";
      let currentStyleKey: string | number = overrides[0] ?? 0;

      for (let i = 0; i < characters.length; i++) {
        const char = characters[i];
        const styleKey: string | number = overrides[i] ?? 0;

        if (styleKey === currentStyleKey) {
          currentRunText += char;
        } else {
          if (currentRunText) {
            runs.push(
              this.createRun(
                currentRunText,
                currentStyleKey,
                styleTable,
                defaultFontFamily,
                defaultFontSize,
                defaultFontWeight,
                defaultItalic,
                defaultUnderline,
                defaultStrike,
                defaultColor,
                scale
              )
            );
          }
          currentRunText = char;
          currentStyleKey = styleKey;
        }
      }

      if (currentRunText) {
        runs.push(
          this.createRun(
            currentRunText,
            currentStyleKey,
            styleTable,
            defaultFontFamily,
            defaultFontSize,
            defaultFontWeight,
            defaultItalic,
            defaultUnderline,
            defaultStrike,
            defaultColor,
            scale
          )
        );
      }
    } else {
      // Single run
      runs.push({
        text: characters,
        fontFamily: defaultFontFamily,
        fontSizePt: defaultFontSize,
        fontWeight: defaultFontWeight,
        italic: defaultItalic,
        underline: defaultUnderline,
        strike: defaultStrike,
        color: defaultColor,
      });
    }

    return runs;
  }

  private createRun(
    text: string,
    styleKey: number | string,
    styleTable: Record<string | number, any>,
    defaultFamily: string,
    defaultSize: number,
    defaultWeight: number,
    defaultItalic: boolean,
    defaultUnderline: boolean,
    defaultStrike: boolean,
    defaultColor: string,
    scale: number
  ): CanvasTextRun {
    const override = styleTable[String(styleKey)];
    if (!override) {
      return {
        text,
        fontFamily: defaultFamily,
        fontSizePt: defaultSize,
        fontWeight: defaultWeight,
        italic: defaultItalic,
        underline: defaultUnderline,
        strike: defaultStrike,
        color: defaultColor,
      };
    }

    let color = defaultColor;
    if (override.fills && override.fills.length > 0) {
      const solid = override.fills.find(
        (f: FigmaPaint) => f.type === "SOLID" && f.visible !== false
      );
      if (solid && solid.color) {
        color = this.figmaColorToHex(solid.color);
      }
    }

    const rawWeight = override.fontWeight;
    const resolvedWeight =
      typeof rawWeight === "number"
        ? rawWeight
        : rawWeight
        ? parseInt(String(rawWeight), 10) || defaultWeight
        : defaultWeight;

    return {
      text,
      fontFamily: override.fontFamily || defaultFamily,
      fontSizePt: override.fontSize ? override.fontSize * scale : defaultSize,
      fontWeight: resolvedWeight,
      italic:
        override.italic !== undefined
          ? override.italic
          : override.fontPostScriptName?.toLowerCase().includes("italic") ||
            defaultItalic,
      underline:
        override.textDecoration === "UNDERLINE" ? true : defaultUnderline,
      strike:
        override.textDecoration === "STRIKETHROUGH" ? true : defaultStrike,
      color,
    };
  }

  private extractTextAlign(node: FigmaNode): "left" | "center" | "right" | "justify" {
    const align = node.style?.textAlignHorizontal;
    if (align === "CENTER") return "center";
    if (align === "RIGHT") return "right";
    if (align === "JUSTIFIED") return "justify";
    return "left";
  }

  private figmaColorToHex(c: FigmaColor): string {
    const r = Math.round((c.r ?? 0) * 255);
    const g = Math.round((c.g ?? 0) * 255);
    const b = Math.round((c.b ?? 0) * 255);
    return (
      "#" +
      [r, g, b]
        .map((x) => x.toString(16).padStart(2, "0"))
        .join("")
        .toUpperCase()
    );
  }

/**
 * Parses XML attributes of an opening or self-closing XML tag into a key-value record,
 * regardless of attribute order or whitespace.
 */
export function parseXmlAttributes(tagStr: string): Record<string, string> {
  const attrs: Record<string, string> = {};
  if (!tagStr) return attrs;
  const attrRegex = /([a-zA-Z0-9_:-]+)\s*=\s*(["'])([\s\S]*?)\2/g;
  let match: RegExpExecArray | null;
  while ((match = attrRegex.exec(tagStr)) !== null) {
    attrs[match[1]] = match[3];
  }
  return attrs;
}

function parseVmlStyle(styleStr: string): Record<string, string> {
  const result: Record<string, string> = {};
  if (!styleStr) return result;
  for (const part of styleStr.split(";")) {
    const idx = part.indexOf(":");
    if (idx !== -1) {
      const k = part.substring(0, idx).trim().toLowerCase();
      const v = part.substring(idx + 1).trim();
      result[k] = v;
    }
  }
  return result;
}

function parsePt(ptStr?: string): number {
  if (!ptStr) return 0;
  const num = parseFloat(ptStr.replace("pt", "").trim());
  return isNaN(num) ? 0 : num;
}

function parseCoordPair(coordStr?: string): [number, number] {
  if (!coordStr) return [0, 0];
  const parts = coordStr.split(",");
  const x = parsePt(parts[0]);
  const y = parsePt(parts[1]);
  return [x, y];
}

export class DocxCanvasParser {
  public async parse(input: Buffer | Uint8Array | ArrayBuffer | string): Promise<CanvasDocument> {
    let buffer: Buffer | Uint8Array | ArrayBuffer;
    if (typeof input === "string") {
      const fs = await import("fs");
      buffer = await fs.promises.readFile(input);
    } else {
      buffer = input;
    }

    const zip = await JSZip.loadAsync(buffer);

    const docXmlFile = zip.file("word/document.xml");
    if (!docXmlFile) {
      throw new Error("Invalid DOCX package: word/document.xml missing.");
    }

    const docXml = await docXmlFile.async("string");
    const warnings: string[] = [];
    const notices: string[] = [];

    // 1. Load relationships for images with attribute-order agnostic XML parsing
    const relsMap = new Map<string, string>(); // rId -> targetPath
    const relsFile = zip.file("word/_rels/document.xml.rels");
    if (relsFile) {
      const relsXml = await relsFile.async("string");
      const relTagRegex = /<Relationship\b[\s\S]*?>/g;
      let rMatch: RegExpExecArray | null;
      while ((rMatch = relTagRegex.exec(relsXml)) !== null) {
        const attrs = parseXmlAttributes(rMatch[0]);
        if (attrs.Id && attrs.Target) {
          // Security & Scope: Skip external relationships
          if (attrs.TargetMode === "External") {
            notices.push(`Skipped external relationship: ${attrs.Id} -> ${attrs.Target}`);
            continue;
          }
          // Sanitize target path against directory traversal
          const cleanTarget = attrs.Target
            .replace(/\\/g, "/")
            .replace(/^(\.\.\/)+/, "")
            .replace(/^(\.\/)+/, "");
          relsMap.set(attrs.Id, cleanTarget);
        }
      }
    }

    // 2. Load media resources
    const resources: Record<string, CanvasResource> = {};
    for (const [rId, target] of relsMap.entries()) {
      if (target.includes("media/")) {
        const zipPath = target.startsWith("word/") ? target : `word/${target}`;
        const imgFile = zip.file(zipPath);
        if (imgFile) {
          const imgData = await imgFile.async("uint8array");
          const resId = target.split("/").pop() || target;
          const extIdx = resId.lastIndexOf(".");
          const ext = extIdx !== -1 ? resId.substring(extIdx).toLowerCase() : "";
          let mimeType = "image/png";
          if (ext === ".jpg" || ext === ".jpeg") mimeType = "image/jpeg";
          else if (ext === ".gif") mimeType = "image/gif";
          else if (ext === ".svg") mimeType = "image/svg+xml";

          resources[resId] = {
            id: resId,
            name: resId,
            mimeType,
            data: imgData,
          };
        }
      }
    }

    // 3. Split paragraphs in w:body using depth tracking to avoid inner textbox paragraphs
    const pages: CanvasPage[] = [];
    const bodyMatch = docXml.match(/<w:body[\s>]([\s\S]*?)<\/w:body>/);
    const bodyContent = bodyMatch ? bodyMatch[1] : docXml;

    // Check for unsupported elements in DOCX body (Scope Definition)
    if (bodyContent.includes("<w:tbl")) {
      warnings.push("Native Word table (<w:tbl>) detected in DOCX body. VML Canvas engine specializes in absolute positioned canvas layouts; native flow tables are not parsed into canvas elements.");
    }
    if (bodyContent.includes("<w:drawing") || bodyContent.includes("<wp:anchor") || bodyContent.includes("<wp:inline")) {
      warnings.push("DrawingML (<w:drawing>) detected in DOCX body. VML Canvas engine specializes in absolute VML shapes (<v:*>); DrawingML shapes were skipped.");
    }

    const paragraphs: string[] = [];
    let pDepth = 0;
    let pStartIndex = -1;
    const tagRegex = /<\/?w:p(?:\s[^>]*?)?(\/?)>/g;
    let tMatch: RegExpExecArray | null;

    while ((tMatch = tagRegex.exec(bodyContent)) !== null) {
      const fullTag = tMatch[0];
      const isSelfClosing = fullTag.endsWith("/>") || tMatch[1] === "/";
      const isClosing = fullTag.startsWith("</");

      if (!isClosing && !isSelfClosing) {
        if (pDepth === 0) {
          pStartIndex = tMatch.index;
        }
        pDepth++;
      } else if (isClosing) {
        pDepth--;
        if (pDepth === 0 && pStartIndex !== -1) {
          paragraphs.push(
            bodyContent.substring(pStartIndex, tMatch.index + fullTag.length)
          );
          pStartIndex = -1;
        }
      } else if (isSelfClosing && pDepth === 0) {
        paragraphs.push(tMatch[0]);
      }
    }

    let pageNumber = 1;
    let currentElements: CanvasElement[] = [];

    const flushPage = () => {
      if (currentElements.length > 0 || pages.length === 0) {
        pages.push({
          pageNumber,
          bounds: { x: 0, y: 0, width: 595.28, height: 841.89 },
          elements: [...currentElements].sort((a, b) => a.zIndex - b.zIndex),
        });
        currentElements = [];
        pageNumber++;
      }
    };

    for (let pIdx = 0; pIdx < paragraphs.length; pIdx++) {
      const pStr = paragraphs[pIdx];

      // Check if this paragraph starts a new page
      const hasPageBreak =
        pStr.includes("<w:pageBreakBefore/>") ||
        (pStr.includes("<w:pageBreakBefore") &&
          !pStr.includes('<w:pageBreakBefore w:val="0"'));
      if (hasPageBreak && pIdx > 0) {
        flushPage();
      }

      // Check for raw top-level body text outside of <w:pict>
      const hasPict = pStr.includes("<w:pict");
      const hasDirectText = pStr.includes("<w:t>") || pStr.includes("<w:t ");
      if (hasDirectText && !hasPict) {
        warnings.push(`Page ${pageNumber}: Top-level paragraph contains flow text outside <w:pict>. Only VML canvas textboxes are converted into canvas elements.`);
      }

      // Extract all pict shapes inside this paragraph (self-closing or paired tags)
      const shapeRegex =
        /<v:(rect|roundrect|shape|line|oval|polyline)\b[^>]*\/>|<v:(rect|roundrect|shape|line|oval|polyline)\b[\s\S]*?<\/v:\2>/g;
      const shapes = Array.from(pStr.matchAll(shapeRegex)).map((m) => m[0]);

      for (const sStr of shapes) {
        // Skip background rects (id="bg_N")
        if (sStr.includes('id="bg_')) continue;

        const elem = this.parseVmlShape(sStr, relsMap, resources, pageNumber, currentElements.length);
        if (elem) {
          currentElements.push(elem);
        }
      }
    }

    // Flush last page
    flushPage();

    return {
      title: "Extracted Canvas Document",
      pageSize: {
        name: "A4",
        widthPt: 595.28,
        heightPt: 841.89,
        unit: "pt",
      },
      marginsPt: { top: 0, right: 0, bottom: 0, left: 0 },
      pages,
      resources,
      diagnostics: { notices, warnings },
    };
  }

  private parseVmlShape(
    shapeXml: string,
    relsMap: Map<string, string>,
    resources: Record<string, CanvasResource>,
    pageIdx: number,
    elemIdx: number
  ): CanvasElement | null {
    const tagMatch = shapeXml.match(/<v:(rect|roundrect|shape|line|oval|polyline)\b/);
    if (!tagMatch) return null;
    const tag = tagMatch[1];

    const openTagMatch = shapeXml.match(/<v:[a-zA-Z0-9_:-]+[^>]*>/);
    const openTagStr = openTagMatch ? openTagMatch[0] : shapeXml;
    const attrs = parseXmlAttributes(openTagStr);
    const styleStr = attrs["style"] || "";
    const styleMap = parseVmlStyle(styleStr);

    // Extract metadata & original IDs preserved in alt / o:title / id
    const altText = attrs["alt"];
    const titleText = attrs["o:title"];
    const elementId = altText || attrs["id"] || `elem_p${pageIdx}_e${elemIdx}`;
    const elementName = titleText || attrs["name"] || undefined;

    const zIndex = parseInt(styleMap["z-index"] || "1", 10);

    // 1. Line shape (<v:line from="..." to="..."/>)
    if (tag === "line") {
      const [x1, y1] = parseCoordPair(attrs["from"]);
      const [x2, y2] = parseCoordPair(attrs["to"]);

      const x = Math.min(x1, x2);
      const y = Math.min(y1, y2);
      const width = Math.abs(x2 - x1);
      const height = Math.abs(y2 - y1);

      const strokeColor = attrs["strokecolor"] ? attrs["strokecolor"] : undefined;
      const strokeWidth = attrs["strokeweight"] ? parsePt(attrs["strokeweight"]) : 1;

      return {
        id: elementId,
        name: elementName || "Line",
        type: "line",
        bounds: { x, y, width, height },
        zIndex,
        style: {
          strokeColor,
          strokeWidth,
        },
      };
    }

    // Coordinates for Rect / Shape / Roundrect
    const x = parsePt(styleMap["margin-left"]);
    const y = parsePt(styleMap["margin-top"]);
    const width = parsePt(styleMap["width"]);
    const height = parsePt(styleMap["height"]);

    const filled = attrs["filled"] !== "f" && attrs["filled"] !== "false";
    const stroked = attrs["stroked"] !== "f" && attrs["stroked"] !== "false";

    const fillColor = filled && attrs["fillcolor"] ? attrs["fillcolor"] : undefined;
    const strokeColor = stroked && attrs["strokecolor"] ? attrs["strokecolor"] : undefined;
    const strokeWidth = stroked && attrs["strokeweight"] ? parsePt(attrs["strokeweight"]) : undefined;

    let cornerRadius: number | undefined;
    if (tag === "roundrect" && attrs["arcsize"]) {
      const arc = parseFloat(attrs["arcsize"]);
      if (!isNaN(arc) && arc > 0) {
        cornerRadius = arc * Math.max(width, height);
      }
    }

    const style: CanvasStyle = {
      fillColor,
      strokeColor,
      strokeWidth,
      cornerRadius,
    };

    // 2. Check imagedata
    const imgTagMatch = shapeXml.match(/<v:imagedata\b[\s\S]*?>/);
    if (imgTagMatch) {
      const imgAttrs = parseXmlAttributes(imgTagMatch[0]);
      const rId = imgAttrs["r:id"] || imgAttrs["id"];
      if (rId) {
        const target = relsMap.get(rId);
        const resId = target ? (target.split("/").pop() || target) : rId;
        return {
          id: elementId,
          name: elementName || imgAttrs["o:title"] || "image",
          type: "image",
          bounds: { x, y, width, height },
          zIndex,
          style,
          imageRef: resId,
        };
      }
    }

    // 3. Check textbox & padding
    let innerPaddingPt: CanvasPadding | undefined;
    const textboxTagMatch = shapeXml.match(/<v:textbox\b[\s\S]*?>/);
    if (textboxTagMatch) {
      const tbAttrs = parseXmlAttributes(textboxTagMatch[0]);
      if (tbAttrs["inset"]) {
        const parts = tbAttrs["inset"].split(",").map((p) => parsePt(p));
        if (parts.length >= 4) {
          innerPaddingPt = {
            left: parts[0],
            top: parts[1],
            right: parts[2],
            bottom: parts[3],
          };
        }
      }
    }

    const textRuns: CanvasTextRun[] = [];
    let align: "left" | "center" | "right" | "justify" = "left";

    const jcTagMatch = shapeXml.match(/<w:jc\b[\s\S]*?>/);
    if (jcTagMatch) {
      const jcAttrs = parseXmlAttributes(jcTagMatch[0]);
      const jc = jcAttrs["w:val"];
      if (jc === "center" || jc === "right" || jc === "justify") {
        align = jc;
      }
    }

    // Check line spacing
    let lineHeightPt: number | undefined;
    const spacingTagMatch = shapeXml.match(/<w:spacing\b[\s\S]*?>/);
    if (spacingTagMatch) {
      const spAttrs = parseXmlAttributes(spacingTagMatch[0]);
      if (spAttrs["w:line"] && spAttrs["w:lineRule"] === "exact") {
        lineHeightPt = parseInt(spAttrs["w:line"], 10) / 20;
      }
    }

    const rRegex = /<w:r[\s>][\s\S]*?<\/w:r>/g;
    const runs = Array.from(shapeXml.matchAll(rRegex)).map((m) => m[0]);

    for (const rXml of runs) {
      const tMatch = rXml.match(/<w:t[^>]*>([\s\S]*?)<\/w:t>/);
      if (tMatch) {
        const text = tMatch[1]
          .replace(/&amp;/g, "&")
          .replace(/&lt;/g, "<")
          .replace(/&gt;/g, ">")
          .replace(/&quot;/g, '"')
          .replace(/&apos;/g, "'");

        const rFontsMatch = rXml.match(/<w:rFonts\b[\s\S]*?>/);
        let fontFamily = "Pretendard";
        if (rFontsMatch) {
          const fAttrs = parseXmlAttributes(rFontsMatch[0]);
          fontFamily = fAttrs["w:eastAsia"] || fAttrs["w:ascii"] || "Pretendard";
        }

        const szMatch = rXml.match(/<w:sz\b[\s\S]*?>/);
        let fontSizePt = 10;
        if (szMatch) {
          const szAttrs = parseXmlAttributes(szMatch[0]);
          if (szAttrs["w:val"]) {
            fontSizePt = parseInt(szAttrs["w:val"], 10) / 2;
          }
        }

        const colorMatch = rXml.match(/<w:color\b[\s\S]*?>/);
        let color = "#000000";
        if (colorMatch) {
          const cAttrs = parseXmlAttributes(colorMatch[0]);
          if (cAttrs["w:val"]) {
            color = `#${cAttrs["w:val"]}`;
          }
        }

        const bold = rXml.includes("<w:b/>") || rXml.includes("<w:b ");
        const italic = rXml.includes("<w:i/>") || rXml.includes("<w:i ");
        const underline = rXml.includes("<w:u ") || rXml.includes("<w:u/>");
        const strike = rXml.includes("<w:strike ") || rXml.includes("<w:strike/>");

        textRuns.push({
          text,
          fontFamily,
          fontSizePt,
          fontWeight: bold ? 700 : 400,
          italic: italic || undefined,
          underline: underline || undefined,
          strike: strike || undefined,
          color,
          lineHeightPt,
        });
      }
    }

    const hasText = textRuns.length > 0;
    let type: CanvasElementType = hasText ? "text" : "card";
    if (tag === "oval") {
      type = "oval";
    } else if (tag === "shape" && attrs["path"]) {
      type = "vector";
    }

    let vectorPoints: { x: number; y: number }[] | undefined;
    let vectorPath: string | undefined;
    if (type === "vector" && attrs["path"]) {
      vectorPath = attrs["path"];
      const cleanPath = vectorPath.replace(/x/g, "z").replace(/e/g, "");
      const parsed = SvgPath.parse(cleanPath);
      const wHwpu = Math.max(10, Math.round(width * 100));
      const hHwpu = Math.max(10, Math.round(height * 100));
      vectorPoints = parsed.toHwpxPolygonPoints(wHwpu, hHwpu);
    }

    return {
      id: elementId,
      name: elementName,
      type,
      bounds: { x, y, width, height },
      zIndex,
      style,
      innerPaddingPt,
      textRuns: hasText ? textRuns : undefined,
      paragraphAlign: align,
      vectorPath,
      vectorPoints,
    };
  }
}

function escapeXml(str: string): string {
  if (!str) return "";
  return str
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;")
    .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, "");
}

function cleanHex(color?: string, fallback: string = "000000"): string {
  if (!color) return fallback;
  return color.replace("#", "").trim().toUpperCase();
}

export function resolveEastAsiaFont(fontFamily?: string): { ascii: string; eastAsia: string } {
  if (!fontFamily) {
    return { ascii: "Pretendard", eastAsia: "맑은 고딕" };
  }
  const f = fontFamily.trim();
  const isKoreanFont =
    f.includes("맑은 고딕") ||
    f.includes("Malgun") ||
    f.includes("Pretendard") ||
    f.includes("Noto Sans KR") ||
    f.includes("Nanum") ||
    f.includes("나눔") ||
    f.includes("돋움") ||
    f.includes("Dotum") ||
    f.includes("굴림") ||
    f.includes("Gulim") ||
    f.includes("바탕") ||
    f.includes("Batang") ||
    f.includes("Gungsuh") ||
    f.includes("KoPub");

  if (isKoreanFont) {
    return { ascii: f, eastAsia: f };
  }
  return { ascii: f, eastAsia: "맑은 고딕" };
}

  public async buildZip(doc: CanvasDocument): Promise<JSZip> {
    const zip = new JSZip();

    // 1. [Content_Types].xml
    zip.file("[Content_Types].xml", this.buildContentTypesXml(doc));

    // 2. _rels/.rels
    zip.file("_rels/.rels", this.buildRootRelsXml());

    // 3. docProps/core.xml & app.xml
    zip.file("docProps/core.xml", this.buildCorePropsXml(doc));
    zip.file("docProps/app.xml", this.buildAppPropsXml(doc));

    // 4. word/styles.xml & settings.xml
    zip.file("word/styles.xml", this.buildStylesXml());
    zip.file("word/settings.xml", this.buildSettingsXml());

    // 5. word/media/
    const imageRelMap = new Map<string, string>(); // resourceId -> rId
    let rIdCounter = 2; // rId1 is styles.xml

    for (const [resId, res] of Object.entries(doc.resources)) {
      const rId = `rId${rIdCounter++}`;
      imageRelMap.set(resId, rId);
      zip.file(`word/media/${resId}`, res.data);
    }

    // 6. word/_rels/document.xml.rels
    zip.file(
      "word/_rels/document.xml.rels",
      this.buildDocumentRelsXml(imageRelMap)
    );

    // 7. word/document.xml
    zip.file("word/document.xml", this.buildDocumentXml(doc, imageRelMap));

    return zip;
  }

  public async renderToUint8Array(doc: CanvasDocument): Promise<Uint8Array> {
    const zip = await this.buildZip(doc);
    return await zip.generateAsync({
      type: "uint8array",
      compression: "DEFLATE",
      compressionOptions: { level: 6 },
    });
  }

  public async renderToBlob(doc: CanvasDocument): Promise<Blob> {
    const uint8 = await this.renderToUint8Array(doc);
    return new Blob([uint8 as any], {
      type: "application/vnd.openxmlformats-officedocument.wordprocessingml.document",
    });
  }

  public async renderToBuffer(doc: CanvasDocument): Promise<Buffer> {
    const uint8 = await this.renderToUint8Array(doc);
    return Buffer.from(uint8.buffer, uint8.byteOffset, uint8.byteLength);
  }

  private buildContentTypesXml(doc: CanvasDocument): string {

    </w:pPrDefault>
  </w:docDefaults>
  <w:style w:type="paragraph" w:default="1" w:styleId="Normal">
    <w:name w:val="Normal"/>
    <w:qFormat/>
  </w:style>
</w:styles>`.trim();
  }

  private buildSettingsXml(): string {
    return `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<w:settings xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main">
  <w:defaultTabStop w:val="720"/>
</w:settings>`.trim();
  }

  private buildDocumentRelsXml(imageRelMap: Map<string, string>): string {
    let rels = `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">
  <Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>
  <Relationship Id="rIdSettings" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/settings" Target="settings.xml"/>`;

    for (const [resId, rId] of imageRelMap.entries()) {
      rels += `\n  <Relationship Id="${rId}" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/image" Target="media/${resId}"/>`;
    }

    rels += "\n</Relationships>";
    return rels.trim();
  }

  private buildDocumentXml(
    doc: CanvasDocument,
    imageRelMap: Map<string, string>
  ): string {
    const widthTwips = ptToTwip(doc.pageSize?.widthPt || 595.28);
    const heightTwips = ptToTwip(doc.pageSize?.heightPt || 841.89);

    const pageParagraphs: string[] = [];

    for (let pIdx = 0; pIdx < doc.pages.length; pIdx++) {
      const page = doc.pages[pIdx];
      const runsXml: string[] = [];

      // 1. If page has custom background color, inject full-page background rect
      if (page.backgroundColor) {
        runsXml.push(
          this.buildBackgroundRect(
            pIdx,
            page.backgroundColor,
            doc.pageSize.widthPt,
            doc.pageSize.heightPt
          )
        );
      }

      // 2. Render all page elements
      for (let eIdx = 0; eIdx < page.elements.length; eIdx++) {
        const elem = page.elements[eIdx];
        const elemXml = this.renderElement(
          pIdx,
          eIdx,
          elem,
          imageRelMap,
          doc.resources
        );
        if (elemXml) {
          runsXml.push(elemXml);
        }
      }

      // Page break & paragraph properties
      const pageBreakBeforeXml =
        pIdx > 0 ? "<w:pageBreakBefore/>" : '<w:pageBreakBefore w:val="0"/>';

      const pPrXml = `<w:pPr>
  ${pageBreakBeforeXml}
  <w:spacing w:before="0" w:after="0" w:line="20" w:lineRule="exact"/>
  <w:autoSpaceDE w:val="0"/>
  <w:autoSpaceDN w:val="0"/>
</w:pPr>`;

      pageParagraphs.push(`<w:p>
  ${pPrXml}
  ${runsXml.join("\n  ")}
</w:p>`);
    }

    // Section properties (A4, 0 margins)
    const sectPrXml = `<w:sectPr>
  <w:pgSz w:w="${widthTwips}" w:h="${heightTwips}" w:orient="portrait"/>
  <w:pgMar w:top="0" w:right="0" w:bottom="0" w:left="0" w:header="0" w:footer="0" w:gutter="0"/>
</w:sectPr>`;

    return `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<w:document xmlns:w="http://schemas.openxmlformats.org/wordprocessingml/2006/main"
            xmlns:v="urn:schemas-microsoft-com:vml"
            xmlns:o="urn:schemas-microsoft-com:office:office"
            xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"
            xmlns:w10="urn:schemas-microsoft-com:office:word"
            xmlns:wp="http://schemas.openxmlformats.org/drawingml/2006/wordprocessingDrawing">
  <w:body>
    ${pageParagraphs.join("\n")}
    ${sectPrXml}
  </w:body>
</w:document>`.trim();
  }

  private buildBackgroundRect(
    pageIdx: number,
    colorHex: string,
    widthPt: number,
    heightPt: number
  ): string {
    const hex = cleanHex(colorHex, "FFFFFF");
    return `<w:r>
  <w:pict>
    <v:rect id="bg_${pageIdx}"
            style="position:absolute;left:0;top:0;margin-left:0pt;margin-top:0pt;width:${widthPt.toFixed(
              3
            )}pt;height:${heightPt.toFixed(
      3
    )}pt;z-index:0;mso-position-horizontal-relative:page;mso-position-vertical-relative:page;mso-wrap-style:none"
            filled="t"
            stroked="f"
            fillcolor="#${hex}"/>
  </w:pict>
</w:r>`;
  }

  private renderElement(
    pageIdx: number,
    elemIdx: number,
    elem: CanvasElement,
    imageRelMap: Map<string, string>,
    resources?: Record<string, CanvasResource>
  ): string {
    const shapeId = `s_${pageIdx}_${elemIdx}`;
    const { x, y, width, height } = elem.bounds;
    const z = elem.zIndex;

    const baseStyle = `position:absolute;left:0;top:0;margin-left:${x.toFixed(
      3
    )}pt;margin-top:${y.toFixed(3)}pt;width:${width.toFixed(
      3
    )}pt;height:${height.toFixed(
      3
    )}pt;z-index:${z};mso-position-horizontal-relative:page;mso-position-vertical-relative:page;mso-wrap-style:none`;

    const altAttr = elem.id ? `alt="${escapeXml(elem.id)}"` : "";
    const titleAttr = `o:title="${escapeXml(elem.name || elem.id || "")}"`;

    // 1. Line
    if (elem.type === "line") {
      const strokeColor = cleanHex(elem.style.strokeColor, "000000");
      const strokeW = elem.style.strokeWidth ? elem.style.strokeWidth.toFixed(2) : "1.00";
      return `<w:r>
  <w:pict>
    <v:line id="${shapeId}"
            ${altAttr}
            ${titleAttr}
            from="${x.toFixed(3)}pt,${y.toFixed(3)}pt"
            to="${(x + width).toFixed(3)}pt,${(y + height).toFixed(3)}pt"
            style="position:absolute;z-index:${z};mso-position-horizontal-relative:page;mso-position-vertical-relative:page"
            strokecolor="#${strokeColor}"
            strokeweight="${strokeW}pt"/>
  </w:pict>
</w:r>`;
    }

    // 1b. Vector Shape
    if (elem.type === "vector") {
      const hasFill = !!elem.style.fillColor;
      const hasStroke = !!elem.style.strokeColor && (elem.style.strokeWidth || 0) > 0;
      const filledAttr = hasFill ? 'filled="t"' : 'filled="f"';
      const fillcolorAttr = hasFill ? `fillcolor="#${cleanHex(elem.style.fillColor)}"` : "";
      const strokedAttr = hasStroke ? 'stroked="t"' : 'stroked="f"';
      const strokecolorAttr = hasStroke ? `strokecolor="#${cleanHex(elem.style.strokeColor)}"` : "";
      const strokeweightAttr = hasStroke
        ? `strokeweight="${(elem.style.strokeWidth || 1).toFixed(2)}pt"`
        : "";

      const pathStr = elem.vectorPath || "";
      if (!pathStr) {
        return "";
      }

      return `<w:r>
  <w:pict>
    <v:shape id="${shapeId}"
             ${altAttr}
             ${titleAttr}
             style="${baseStyle}"
             coordsize="1000,1000"
             path="${pathStr}"
             ${filledAttr}
             ${fillcolorAttr}
             ${strokedAttr}
             ${strokecolorAttr}
             ${strokeweightAttr}/>
  </w:pict>
</w:r>`;
    }

    // 2. Image
    if (elem.type === "image" && elem.imageRef) {
      const relId = imageRelMap.get(elem.imageRef) || "rId1";
      let cropAttr = "";
      const res = resources && elem.imageRef ? resources[elem.imageRef] : undefined;
      if (res && res.data && res.data.length > 24) {
        const d = res.data;
        const readU32 = (off: number) =>
          ((d[off] << 24) >>> 0) + (d[off + 1] << 16) + (d[off + 2] << 8) + d[off + 3];
        const readU16 = (off: number) =>
          (d[off] << 8) | d[off + 1];

        let imgW = 0;
        let imgH = 0;
        // PNG check: 89 50 4E 47
        if (
          d[0] === 0x89 &&
          d[1] === 0x50 &&
          d[2] === 0x4e &&
          d[3] === 0x47
        ) {
          imgW = readU32(16);
          imgH = readU32(20);
        } else if (d[0] === 0xff && d[1] === 0xd8) {
          // Basic JPEG SOF0/SOF2 marker parser
          let offset = 2;
          while (offset < d.length - 8) {
            const marker = readU16(offset);
            offset += 2;
            if (marker === 0xffc0 || marker === 0xffc2) {
              imgH = readU16(offset + 3);
              imgW = readU16(offset + 5);
              break;
            }
            const len = readU16(offset);
            offset += len;
          }
        }

        if (imgW > 0 && imgH > 0 && width > 0 && height > 0) {
          const arImg = imgW / imgH;
          const arBox = width / height;
          if (arImg < arBox * 0.99) {
            // Image is taller than container box: crop top & bottom (object-fit: cover)
            const cropY = (1.0 - arImg / arBox) / 2.0;
            const cropVal = Math.round(cropY * 65536);
            cropAttr = ` croptop="${cropVal}f" cropbottom="${cropVal}f"`;
          } else if (arImg > arBox * 1.01) {
            // Image is wider than container box: crop left & right
            const cropX = (1.0 - arBox / arImg) / 2.0;
            const cropVal = Math.round(cropX * 65536);
            cropAttr = ` cropleft="${cropVal}f" cropright="${cropVal}f"`;
          }
        }
      }

      return `<w:r>
  <w:pict>
    <v:shape id="${shapeId}"
             ${altAttr}
             ${titleAttr}
             style="${baseStyle}"
             filled="f"
             stroked="f">
      <v:imagedata r:id="${relId}"${cropAttr} o:title="${escapeXml(elem.name || "image")}"/>
    </v:shape>
  </w:pict>
</w:r>`;
    }

    // 3. Card or Text
    const isText = elem.type === "text" || (elem.textRuns && elem.textRuns.length > 0);
    const hasFill = !!elem.style.fillColor;
    const hasStroke = !!elem.style.strokeColor && (elem.style.strokeWidth || 0) > 0;

    const filledAttr = hasFill ? 'filled="t"' : 'filled="f"';
    const fillcolorAttr = hasFill ? `fillcolor="#${cleanHex(elem.style.fillColor)}"` : "";

    const strokedAttr = hasStroke ? 'stroked="t"' : 'stroked="f"';
    const strokecolorAttr = hasStroke ? `strokecolor="#${cleanHex(elem.style.strokeColor)}"` : "";
    const strokeweightAttr = hasStroke
      ? `strokeweight="${(elem.style.strokeWidth || 1).toFixed(2)}pt"`
      : "";

    // Corner radius or Oval
    const isOval = elem.type === "oval";
    const isRound = !isOval && !!elem.style.cornerRadius && elem.style.cornerRadius > 0;
    const tag = isOval ? "v:oval" : isRound ? "v:roundrect" : "v:rect";
    const arcsizeAttr = isRound
      ? `arcsize="${Math.min(
          0.5,
          (elem.style.cornerRadius || 0) / Math.max(width, height)
        ).toFixed(3)}"`
      : "";

    let textboxContentXml = "";
    if (isText && elem.textRuns) {
      const jcVal = elem.paragraphAlign || "left";
      const runsXml: string[] = [];

      for (const run of elem.textRuns) {
        const { ascii, eastAsia } = resolveEastAsiaFont(run.fontFamily);
        const fontAscii = escapeXml(ascii);
        const fontEastAsia = escapeXml(eastAsia);
        const fontSizeHalfPt = Math.round((run.fontSizePt || 10) * 2);
        const colorHex = cleanHex(run.color, "000000");

        const boldXml = (run.fontWeight || 400) >= 600 ? "<w:b/>" : "";
        const italicXml = run.italic ? "<w:i/>" : "";
        const uXml = run.underline ? '<w:u w:val="single"/>' : "";
        const strikeXml = run.strike ? "<w:strike/>" : "";

        runsXml.push(`<w:r>
  <w:rPr>
    <w:rFonts w:ascii="${fontAscii}" w:hAnsi="${fontAscii}" w:eastAsia="${fontEastAsia}" w:cs="${fontAscii}"/>
    <w:sz w:val="${fontSizeHalfPt}"/>
    <w:szCs w:val="${fontSizeHalfPt}"/>
    <w:color w:val="${colorHex}"/>
    ${boldXml}
    ${italicXml}
    ${uXml}
    ${strikeXml}
  </w:rPr>
  <w:t xml:space="preserve">${escapeXml(run.text)}</w:t>
</w:r>`);
      }

      const pad = elem.innerPaddingPt || {};
      const padTop = pad.top ?? 0;
      const padRight = pad.right ?? 0;
      const padBottom = pad.bottom ?? 0;
      const padLeft = pad.left ?? 0;
      const insetAttr = `inset="${padLeft.toFixed(2)}pt,${padTop.toFixed(2)}pt,${padRight.toFixed(2)}pt,${padBottom.toFixed(2)}pt"`;

      const firstLineHeight = elem.textRuns.find((r) => r.lineHeightPt)?.lineHeightPt;
      const spacingXml = firstLineHeight
        ? `<w:spacing w:before="0" w:after="0" w:line="${Math.round(firstLineHeight * 20)}" w:lineRule="exact"/>`
        : `<w:spacing w:before="0" w:after="0" w:line="240" w:lineRule="auto"/>`;

      textboxContentXml = `
      <v:textbox style="mso-fit-shape-to-text:f;mso-next-textbox:none" ${insetAttr}>
        <w:txbxContent>
          <w:p>
            <w:pPr>
              <w:jc w:val="${jcVal}"/>
              ${spacingXml}
            </w:pPr>
            ${runsXml.join("\n            ")}
          </w:p>
        </w:txbxContent>
      </v:textbox>`;
    }

    return `<w:r>
  <w:pict>
    <${tag} id="${shapeId}"
            ${altAttr}
            ${titleAttr}
            style="${baseStyle}"
            ${filledAttr}
            ${fillcolorAttr}
            ${strokedAttr}
            ${strokecolorAttr}
            ${strokeweightAttr}
            ${arcsizeAttr}>${textboxContentXml}
    </${tag}>
  </w:pict>
</w:r>`;
  }
}

import JSZip from "jszip";
import {
  CanvasDocument,

export interface HwpxCanvasRendererOptions {
  creator?: string;

function escapeXml(str: string): string {
  if (!str) return "";
  return str
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&apos;")
    .replace(/[\x00-\x08\x0B\x0C\x0E-\x1F]/g, "");
}

function cleanHex(color?: string, fallback: string = "000000"): string {
  if (!color) return fallback;
  return color.replace("#", "").trim().toUpperCase();
}

interface CharPrKey {
  fontId: number;
  fontSizePt: number;
  colorHex: string;
  bold: boolean;
  italic: boolean;
}

export class HwpxCanvasRenderer {
  private creator: string;

  constructor(options?: HwpxCanvasRendererOptions) {
    this.creator = options?.creator || "Figma2Word Canvas Engine";
  }

  public async buildZip(doc: CanvasDocument): Promise<JSZip> {
    const zip = new JSZip();

    // 1. mimetype (MUST be first entry, STORED uncompressed, 19 bytes)
    zip.file("mimetype", "application/hwp+zip", {
      compression: "STORE",
    });

    // 2. version.xml & settings.xml
    zip.file("version.xml", this.buildVersionXml());
    zip.file("settings.xml", this.buildSettingsXml());

    // 3. META-INF/container.xml
    zip.file("META-INF/container.xml", this.buildContainerXml());

    // 4. BinData/
    const imageBinMap = new Map<string, string>(); // resourceId -> BIN0001
    let binCounter = 1;

    for (const [resId, res] of Object.entries(doc.resources)) {
      const binId = `BIN${binCounter.toString().padStart(4, "0")}`;
      const ext = res.mimeType === "image/jpeg" ? "jpg" : "png";
      const binFileName = `${binId}.${ext}`;
      imageBinMap.set(resId, binId);
      zip.file(`BinData/${binFileName}`, res.data);
      binCounter++;
    }

    // 5. META-INF/manifest.xml & Contents/content.hpf
    zip.file("META-INF/manifest.xml", this.buildManifestXml(imageBinMap));
    zip.file("Contents/content.hpf", this.buildContentHpf(doc, imageBinMap));

    // 6. Dynamic Font Registry (7 OWPML languages)
    const fontRegistry = new Map<string, number>();
    fontRegistry.set("맑은 고딕", 0); // Default fallback at index 0

    for (const page of doc.pages) {
      for (const elem of page.elements) {
        if (elem.textRuns) {
          for (const run of elem.textRuns) {
            const font = run.fontFamily?.trim() || "맑은 고딕";
            if (!fontRegistry.has(font)) {
              fontRegistry.set(font, fontRegistry.size);
            }
          }
        }
      }
    }

    // Dynamic style registries
    const charPrRegistry = new Map<string, number>(); // serializedKey -> id
    const charPrList: { id: number; key: CharPrKey }[] = [];

    const getCharPrId = (run: CanvasTextRun): number => {
      const font = run.fontFamily?.trim() || "맑은 고딕";
      const fontId = fontRegistry.get(font) ?? 0;
      const key: CharPrKey = {
        fontId,
        fontSizePt: Math.round((run.fontSizePt || 10) * 10) / 10,
        colorHex: cleanHex(run.color, "000000"),
        bold: (run.fontWeight || 400) >= 600,
        italic: !!run.italic,
      };
      const serialized = `${fontId}_${key.fontSizePt}_${key.colorHex}_${key.bold}_${key.italic}`;
      if (charPrRegistry.has(serialized)) {
        return charPrRegistry.get(serialized)!;
      }
      const newId = charPrList.length;
      charPrRegistry.set(serialized, newId);
      charPrList.push({ id: newId, key });
      return newId;
    };

    // 7. Contents/section0.xml (builds shapes & populates charPrRegistry)
    const section0Xml = this.buildSection0Xml(doc, imageBinMap, getCharPrId);
    zip.file("Contents/section0.xml", section0Xml);

    // 8. Contents/header.xml (outputs registered fontfaces & charPr styles)
    zip.file("Contents/header.xml", this.buildHeaderXml(charPrList, fontRegistry));

    return zip;
  }

  public async renderToUint8Array(doc: CanvasDocument): Promise<Uint8Array> {
    const zip = await this.buildZip(doc);
    return await zip.generateAsync({
      type: "uint8array",
      compression: "DEFLATE",
      compressionOptions: { level: 6 },
    });
  }

  public async renderToBlob(doc: CanvasDocument): Promise<Blob> {
    const uint8 = await this.renderToUint8Array(doc);
    return new Blob([uint8 as any], {
      type: "application/hwp+zip",
    });
  }

  public async renderToBuffer(doc: CanvasDocument): Promise<Buffer> {
    const uint8 = await this.renderToUint8Array(doc);
    return Buffer.from(uint8.buffer, uint8.byteOffset, uint8.byteLength);
  }

  private buildHeaderXml(
    charPrList: { id: number; key: CharPrKey }[],
    fontRegistry: Map<string, number>
  ): string {
    let charPrEntries = "";
    for (const { id, key } of charPrList) {
      const heightHwpu = Math.round(key.fontSizePt * 100);
      const boldAttr = key.bold ? ' bold="1"' : "";
      const italicAttr = key.italic ? ' italic="1"' : "";
      charPrEntries += `\n    <hh:charPr id="${id}" height="${heightHwpu}" textColor="#${key.colorHex}" shadeColor="none" useFontSpace="1" useKerning="1" symMark="NONE" borderFillIDRef="1"${boldAttr}${italicAttr}>
      <hh:fontRef hangul="${key.fontId}" latin="${key.fontId}" hanja="${key.fontId}" japanese="${key.fontId}" other="${key.fontId}" symbol="${key.fontId}" user="${key.fontId}"/>
      <hh:ratio hangul="100" latin="100" hanja="100" japanese="100" other="100" symbol="100" user="100"/>
      <hh:spacing hangul="0" latin="0" hanja="0" japanese="0" other="0" symbol="0" user="0"/>
      <hh:relSz hangul="100" latin="100" hanja="100" japanese="100" other="100" symbol="100" user="100"/>
      <hh:offset hangul="0" latin="0" hanja="0" japanese="0" other="0" symbol="0" user="0"/>
    </hh:charPr>`;
    }

    if (charPrList.length === 0) {
      charPrEntries = `\n    <hh:charPr id="0" height="1000" textColor="#000000" shadeColor="none" useFontSpace="1" useKerning="1" symMark="NONE" borderFillIDRef="1">
      <hh:fontRef hangul="0" latin="0" hanja="0" japanese="0" other="0" symbol="0" user="0"/>
    </hh:charPr>`;
    }

    const langs = ["hangul", "latin", "hanja", "japanese", "other", "symbol", "user"];
    const fontCnt = fontRegistry.size;
    const fontXmlEntries = Array.from(fontRegistry.entries())
      .map(([face, id]) => `<hh:font id="${id}" face="${escapeXml(face)}" type="TTF" isEmbedded="0"/>`)
      .join("\n        ");

    const fontfaceBlocks = langs

  private buildSection0Xml(
    doc: CanvasDocument,
    imageBinMap: Map<string, string>,
    getCharPrId: (run: CanvasTextRun) => number
  ): string {
    const pageParas: string[] = [];
    let globalShapeId = 1;

    for (let pIdx = 0; pIdx < doc.pages.length; pIdx++) {
      const page = doc.pages[pIdx];
      const pageBreakAttr = pIdx === 0 ? ' pageBreak="0"' : ' pageBreak="1"';

      const shapesXml: string[] = [];

      // 1. Full-page background rect if specified
      if (page.backgroundColor) {
        const bgW = ptToHwpu(doc.pageSize?.widthPt || 595.28);
        const bgH = ptToHwpu(doc.pageSize?.heightPt || 841.89);
        shapesXml.push(
          `<hp:run><hp:rect id="${globalShapeId++}" zOrder="0" numberingType="NONE" textWrap="IN_FRONT_OF_TEXT" textFlow="BOTH_SIDES" lock="0" dropcapstyle="None" href="" groupLevel="0" instid="${globalShapeId}" ratio="0">
  <hp:offset x="0" y="0"/>
  <hp:orgSz width="${bgW}" height="${bgH}"/>
  <hp:curSz width="${bgW}" height="${bgH}"/>
  <hp:flip horizontal="0" vertical="0"/>
  <hp:rotationInfo angle="0" centerX="${Math.round(bgW / 2)}" centerY="${Math.round(bgH / 2)}" rotateimage="1"/>
  <hp:renderingInfo>
    <hc:transMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:scaMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:rotMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
  </hp:renderingInfo>
  <hp:pos treatAsChar="0" horzRelTo="PAPER" vertRelTo="PAPER" horzAlign="LEFT" vertAlign="TOP" horzOffset="0" vertOffset="0"/>
  <hp:lineShape color="#000000" width="0" style="NONE" endCap="FLAT" headStyle="NORMAL" tailStyle="NORMAL" headfill="0" tailfill="0" headSz="SMALL_SMALL" tailSz="SMALL_SMALL" outlineStyle="NORMAL" alpha="0"/>
  <hc:fillBrush><hc:winBrush faceColor="#${cleanHex(page.backgroundColor)}" hatchColor="#999999" alpha="0"/></hc:fillBrush>
</hp:rect></hp:run>`
        );
      }

      // 2. Render all elements
      for (const elem of page.elements) {
        const rendered = this.renderElement(
          globalShapeId++,
          elem,
          imageBinMap,
          getCharPrId
        );
        if (rendered) {
          shapesXml.push(`<hp:run>${rendered}</hp:run>`);
        }
      }

      // First page includes secPr
      const secPrXml =
        pIdx === 0
          ? `<hp:secPr id="0" textDirection="HORIZONTAL" spaceColumns="1134" tabStop="8000" outlineShapeIDRef="0" memoShapeIDRef="0" textVerticalWidthHead="0">

    <hp:margin header="0" footer="0" gutter="0" left="0" right="0" top="0" bottom="0"/>
  </hp:pagePr>
</hp:secPr>`
          : "";

      pageParas.push(`<hp:p id="${pIdx}" paraPrIDRef="0" styleIDRef="0"${pageBreakAttr} columnBreak="0" merged="0">
  ${secPrXml}
  ${shapesXml.join("\n  ")}

    return `<?xml version="1.0" encoding="UTF-8" standalone="yes"?>

  private renderElement(
    shapeId: number,
    elem: CanvasElement,
    imageBinMap: Map<string, string>,
    getCharPrId: (run: CanvasTextRun) => number
  ): string {
    const xHwpu = ptToHwpu(elem.bounds.x);
    const yHwpu = ptToHwpu(elem.bounds.y);
    const wHwpu = Math.max(100, ptToHwpu(elem.bounds.width));
    const hHwpu = Math.max(100, ptToHwpu(elem.bounds.height));
    const zOrder = elem.zIndex;

    // 1. Image
    if (elem.type === "image" && elem.imageRef) {
      const binId = imageBinMap.get(elem.imageRef);
      if (binId) {
        return `<hp:pic id="${shapeId}" zOrder="${zOrder}" numberingType="NONE" textWrap="IN_FRONT_OF_TEXT" textFlow="BOTH_SIDES" lock="0" dropcapstyle="None" href="" groupLevel="0" instid="${shapeId}" ratio="0">
  <hp:offset x="0" y="0"/>
  <hp:orgSz width="${wHwpu}" height="${hHwpu}"/>
  <hp:curSz width="${wHwpu}" height="${hHwpu}"/>
  <hp:flip horizontal="0" vertical="0"/>
  <hp:rotationInfo angle="0" centerX="${Math.round(wHwpu / 2)}" centerY="${Math.round(hHwpu / 2)}" rotateimage="1"/>
  <hp:renderingInfo>
    <hc:transMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:scaMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:rotMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
  </hp:renderingInfo>
  <hp:pos treatAsChar="0" horzRelTo="PAPER" vertRelTo="PAPER" horzAlign="LEFT" vertAlign="TOP" horzOffset="${xHwpu}" vertOffset="${yHwpu}"/>
  <hp:lineShape color="#000000" width="0" style="NONE" endCap="FLAT" headStyle="NORMAL" tailStyle="NORMAL" headfill="0" tailfill="0" headSz="SMALL_SMALL" tailSz="SMALL_SMALL" outlineStyle="NORMAL" alpha="0"/>
  <hp:imgRect pt0_x="0" pt0_y="0" pt1_x="${wHwpu}" pt1_y="0" pt2_x="${wHwpu}" pt2_y="${hHwpu}" pt3_x="0" pt3_y="${hHwpu}"/>
  <hp:imgClip left="0" right="${wHwpu}" top="0" bottom="${hHwpu}"/>
  <hp:inImgDim dimwidth="${wHwpu}" dimheight="${hHwpu}"/>
  <hc:img binRef="${binId}"/>
</hp:pic>`;
      }
    }

    // 2. Card / Rect / Textbox
    const isText = elem.type === "text" || (elem.textRuns && elem.textRuns.length > 0);
    const hasFill = !!elem.style.fillColor;
    const hasStroke = !!elem.style.strokeColor && (elem.style.strokeWidth || 0) > 0;

    const lineStyle = hasStroke ? "SOLID" : "NONE";
    const lineColor = hasStroke ? `#${cleanHex(elem.style.strokeColor)}` : "#000000";
    const lineWidth = hasStroke ? Math.round((elem.style.strokeWidth || 1) * 10).toString() : "0";

    const fillBrush = hasFill
      ? `<hc:fillBrush><hc:winBrush faceColor="#${cleanHex(elem.style.fillColor)}" hatchColor="#999999" alpha="0"/></hc:fillBrush>`
      : `<hc:fillBrush><hc:winBrush faceColor="none" hatchColor="#999999" alpha="0"/></hc:fillBrush>`;

    const isLine = elem.type === "line";
    const shapeW = isLine ? Math.max(10, wHwpu) : wHwpu;
    const shapeH = isLine ? Math.max(10, hHwpu) : hHwpu;

    let drawTextPart = "";
    if (isText && elem.textRuns && elem.textRuns.length > 0) {
      let paraPrId = 0;
      if (elem.paragraphAlign === "center") paraPrId = 1;
      else if (elem.paragraphAlign === "right") paraPrId = 2;
      else if (elem.paragraphAlign === "justify") paraPrId = 3;

      const runsXml: string[] = [];
      for (const run of elem.textRuns) {
        const charPrId = getCharPrId(run);
        runsXml.push(
          `<hp:run charPrIDRef="${charPrId}"><hp:t>${escapeXml(run.text)}</hp:t></hp:run>`
        );
      }

      const drawName = escapeXml(elem.name || elem.id || "");
      drawTextPart = `
  <hp:drawText lastWidth="${shapeW}" name="${drawName}" editable="1">
    <hp:subList id="" textDirection="HORIZONTAL" lineWrap="SQUEEZE" vertAlign="TOP" linkListIDRef="0" linkListNextIDRef="0" textWidth="${shapeW}" textHeight="${shapeH}" hasTextRef="0" hasNumRef="0">
      <hp:p id="0" paraPrIDRef="${paraPrId}" styleIDRef="0" pageBreak="0" columnBreak="0" merged="0">
        ${runsXml.join("")}
      </hp:p>
    </hp:subList>
  </hp:drawText>`;
    }

    // 3. Vector Shape (KS X 6101 native hp:polygon)
    if (elem.type === "vector") {
      const pts = elem.vectorPoints || [];
      const ptXml =
        pts.length > 0
          ? pts.map((p) => `<hp:pt x="${p.x}" y="${p.y}"/>`).join("\n  ")
          : `<hp:pt x="0" y="0"/>\n  <hp:pt x="${shapeW}" y="0"/>\n  <hp:pt x="${shapeW}" y="${shapeH}"/>\n  <hp:pt x="0" y="${shapeH}"/>`;

      return `<hp:polygon id="${shapeId}" zOrder="${zOrder}" numberingType="NONE" textWrap="IN_FRONT_OF_TEXT" textFlow="BOTH_SIDES" lock="0" dropcapstyle="None" href="" groupLevel="0" instid="${shapeId}" ratio="0">
  <hp:offset x="0" y="0"/>
  <hp:orgSz width="${shapeW}" height="${shapeH}"/>
  <hp:curSz width="${shapeW}" height="${shapeH}"/>
  <hp:flip horizontal="0" vertical="0"/>
  <hp:rotationInfo angle="0" centerX="${Math.round(shapeW / 2)}" centerY="${Math.round(shapeH / 2)}" rotateimage="1"/>
  <hp:renderingInfo>
    <hc:transMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:scaMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:rotMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
  </hp:renderingInfo>
  <hp:pos treatAsChar="0" horzRelTo="PAPER" vertRelTo="PAPER" horzAlign="LEFT" vertAlign="TOP" horzOffset="${xHwpu}" vertOffset="${yHwpu}"/>
  <hp:lineShape color="${lineColor}" width="${lineWidth}" style="${lineStyle}" endCap="FLAT" headStyle="NORMAL" tailStyle="NORMAL" headfill="0" tailfill="0" headSz="SMALL_SMALL" tailSz="SMALL_SMALL" outlineStyle="NORMAL" alpha="0"/>
  ${fillBrush}
  ${ptXml}${drawTextPart}
</hp:polygon>`;
    }

    const shapeTag = elem.type === "oval" ? "hp:oval" : "hp:rect";

    return `<${shapeTag} id="${shapeId}" zOrder="${zOrder}" numberingType="NONE" textWrap="IN_FRONT_OF_TEXT" textFlow="BOTH_SIDES" lock="0" dropcapstyle="None" href="" groupLevel="0" instid="${shapeId}" ratio="0">
  <hp:offset x="0" y="0"/>
  <hp:orgSz width="${shapeW}" height="${shapeH}"/>
  <hp:curSz width="${shapeW}" height="${shapeH}"/>
  <hp:flip horizontal="0" vertical="0"/>
  <hp:rotationInfo angle="0" centerX="${Math.round(shapeW / 2)}" centerY="${Math.round(shapeH / 2)}" rotateimage="1"/>
  <hp:renderingInfo>
    <hc:transMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:scaMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
    <hc:rotMatrix e1="1" e2="0" e3="0" e4="0" e5="1" e6="0"/>
  </hp:renderingInfo>
  <hp:pos treatAsChar="0" horzRelTo="PAPER" vertRelTo="PAPER" horzAlign="LEFT" vertAlign="TOP" horzOffset="${xHwpu}" vertOffset="${yHwpu}"/>
  <hp:lineShape color="${lineColor}" width="${lineWidth}" style="${lineStyle}" endCap="FLAT" headStyle="NORMAL" tailStyle="NORMAL" headfill="0" tailfill="0" headSz="SMALL_SMALL" tailSz="SMALL_SMALL" outlineStyle="NORMAL" alpha="0"/>
  ${fillBrush}${drawTextPart}
</${shapeTag}>`;
  }

function parseCliArgs(args: string[]) {
  const result: {
    url?: string;
    token?: string;
    output?: string;
    orientation?: "portrait" | "landscape";
    engine?: "flow" | "canvas";
    format?: "docx" | "hwpx" | "both";
    scale?: "fit-a4" | "css-96dpi" | "1:1";
    files: string[];
  } = { files: [], engine: "flow", format: "docx" };

  for (let i = 0; i < args.length; i++) {
    const arg = args[i];
    if (arg === "--url" || arg === "-u") {
      result.url = args[++i];
    } else if (arg === "--token" || arg === "-t") {
      result.token = args[++i];
    } else if (arg === "--output" || arg === "-o") {
      result.output = args[++i];
    } else if (arg === "--orientation" || arg === "-r") {
      const val = args[++i]?.toLowerCase();
      if (val === "landscape" || val === "portrait") {
        result.orientation = val;
      }
    } else if (arg === "--engine" || arg === "-e") {
      const val = args[++i]?.toLowerCase();
      if (val === "canvas" || val === "flow") {
        result.engine = val;
      }
    } else if (arg === "--format" || arg === "-f") {
      const val = args[++i]?.toLowerCase();
      if (val === "docx" || val === "hwpx" || val === "both") {
        result.format = val;
      }
    } else if (arg === "--scale" || arg === "-s") {
      const val = args[++i]?.toLowerCase();
      if (val === "fit-a4" || val === "css-96dpi" || val === "1:1") {
        result.scale = val as any;
      }
    } else if (!arg.startsWith("-")) {
      result.files.push(arg);
    }
  }

  return result;
}

# Phase 15: Native Vector Engine Technical Report (문제 A & B 해결)

## 1. 개요 및 배경

Figma 디자인에는 단순 사각형(`RECTANGLE`) 외에도 복잡한 아이콘, 상태 표시 핀, 로고, 별표, 화살표 등 수많은 벡터(`VECTOR`, `ELLIPSE`, `BOOLEAN_OPERATION`) 요소가 포함되어 있다.
기존 엔진에서는 이러한 벡터 요소를 Word와 한글에서 표현할 때 다음과 같은 두 가지 심각한 기술적 결함(Technical Problems A & B)이 존재했다:

1. **Problem A (HWPX 폴리곤 결손)**:
   HWPX 렌더러가 비-직사각형 벡터 요소를 만났을 때, 적절한 OWPML 다각형 태그를 생성하지 못하고 단순 직사각형(`<hp:rect>`)으로 강제 대체(Fallback)하여 시각적 형태가 완전히 파괴되는 문제.
2. **Problem B (SVG Path 파서 및 VML 좌표계 불일치)**:
   Figma AST의 `fillGeometry[].path`는 SVG 경로 문자열(d-string)로 주어지나, 상대 좌표(`m, l, c, s, q, t, a`), 호(`A/a`), 스무스 베지에(`S/s, T/t`) 등 다양한 복합 명령어가 혼재되어 있었다. 이를 Word VML의 정규화된 가상 좌표계(`coordsize="1000,1000"`)로 맵핑하지 못해 벡터가 찌그러지거나 사라지는 문제.

---

## 2. 해결 아키텍처 및 구현 내역

### 2.1 완전한 SVG Path 파서 엔진 (`src/canvas/geometry/svgPathParser.ts`)
본 단계에서 새롭게 설계·구현된 `SvgPath` 파서는 W3C SVG 1.1 / 2.0 표준 경로 명세를 100% 지원한다:
- **지원 명령어**:
  - 이동: `M / m` (절대/상대 이동)
  - 직선: `L / l`, `H / h` (수평), `V / v` (수직)
  - 3차 베지에 곡선: `C / c`, `S / s` (반사 제어점 자동 계산)
  - 2차 베지에 곡선: `Q / q`, `T / t` (3차 베지에로 내부 자동 승격)
  - 타원 호: `A / a` (W3C 중심각 매개변수화 알고리즘을 통한 다중 3차 베지에 세그먼트 변환)
  - 경로 닫기: `Z / z`
- **바운딩 박스 계산 및 좌표 정규화**:
  SVG 경로 내 모든 제어점과 정점의 최소/최대 좌표(`minX, minY, maxX, maxY`)를 계산한 뒤, 가상 좌표계 `[0, 1000] x [0, 1000]`으로 스케일 변환을 수행한다.

### 2.2 Word VML 렌더러 통합 (`docxCanvasRenderer.ts`)
정규화된 경로를 VML 문법에 맞게 직렬화한다:
```xml
<v:shape id="shape_vec_1"
         style="position:absolute;left:55pt;top:410pt;width:32pt;height:32pt;z-index:10"
         coordsize="1000,1000"
         path="m 375,0 c 468,0 718,218 ... x e"
         fillcolor="#F3BF1A"
         stroked="f">
</v:shape>
```
- `coordsize="1000,1000"`을 고정하여 셰이프의 물리적 크기(`width, height`)가 변경되어도 벡터 형태가 비율을 유지하며 선명하게 렌더링됨.

### 2.3 한컴 HWPX OWPML `<hp:polygon>` 렌더러 통합 (`hwpxCanvasRenderer.ts`)
HWPX는 베지에 곡선을 직접 표현하는 전용 태그가 없으므로, 베지에 곡선 구간을 $t \in [0, 1]$ 구간에서 정밀 샘플링하여 HWPU 단위의 다각형 정점 리스트(`<hp:pt>`)로 변환한다:
```xml
<hp:polygon id="101" zOrder="10" numberingType="NONE" textWrap="IN_FRONT_OF_TEXT">
  <hp:offset x="5500" y="41000"/>
  <hp:orgSz width="3200" height="3200"/>
  <hp:curSz width="3200" height="3200"/>
  <hp:lineShape color="#000000" width="0" style="SOLID"/>
  <hp:fillBrush>
    <hp:winBrush faceColor="#F3BF1A" hatchColor="#000000" alpha="0"/>
  </hp:fillBrush>
  <hp:pt x="1200" y="0"/>
  <hp:pt x="1498" y="699"/>
  <hp:pt x="2198" y="700"/>
  <hp:pt x="1648" y="1150"/>
  <!-- ... 정밀 다각형 정점 배열 ... -->
</hp:polygon>
```
이로써 **Problem A(단순 사각형 대체 문제)가 원천적으로 제거**되었으며, 한컴오피스 한글에서 네이티브 벡터 다각형으로 정확히 렌더링된다.

## 4. Technical Problem E: 보고서 2 페이지 8 사실 확인 (Ground Truth)

과거 개발 메모에서 제기되었던 *"보고서 2 페이지 8의 이미지가 4개인가 8개인가, Canvas가 이미지를 누락한 것이 아닌가"*라는 의문에 대해, 3대 원본 데이터를 대조하는 독립 감사(`audit_report2_page8.py`)를 수행하였다.

### 4.1 3각 대조 감사 데이터
1. **Figma AST 원본**: `research/rea-document-engine/reference-corpus/fixtures/Report2-Live-Snapshot.json`
2. **Canvas 생성 DOCX**: `research/rea-document-engine/phase13c/renders/canvas_report2/`
3. **디자이너 수작업 완성본**: `research/rea-document-engine/reference-corpus/docx/완성본-20261012/`

### 4.2 규명된 결정적 진실 (Ground Truth)
1. **이미지 개수 오해의 원인**:
   - 보고서 2의 7페이지(Frame 6: `P6a 결함Top5(1)`)에는 결함 사진이 **4개** 배치되어 있음.
   - 8페이지(Frame 7: `P6b 결함Top5(2)`)에는 결함 사진이 **6개** + 상단 로고 **1개** = **총 7개의 이미지가 배치되어 있음.**
   - 즉, "4 vs 8"이라는 혼란은 7페이지와 8페이지의 구성을 오인하여 발생한 단순 착오였음.
2. **인간 디자이너의 임의 수정과 시각적 차이 (45.62% Pixel Mismatch 원인)**:
   - **Figma 원본**: 6장의 결함 사진이 각각 가로 341px, 세로 255.75px (4:3 비율, 191.8pt)로 배치되어 있으며, 각 사진 상단에 `1차 (육안)`, `2차 (육안)` 배지 텍스트박스가 존재함.
   - **Canvas 엔진**: Figma 원본 설계에 100% 충실하게 191.8pt 높이와 6개의 배지를 온전히 렌더링함.
   - **인간 디자이너 완성본**: 디자이너가 Word 수작업 편집 과정에서 1페이지 안에 여백을 더 확보하기 위해 **6장 사진의 세로 높이를 191.8pt에서 150.0pt로 일괄 41.8pt 압축(Squish)**시켰으며, **6개의 배지 텍스트박스를 수작업으로 삭제(Omit)**하였음.
   - 행당 41.8pt씩 총 125pt의 수직 위치 밀림이 발생하여 이미지 픽셀 차이가 발생한 것이었음.

### 4.3 결론
Canvas 엔진은 피그마 원본 디자인을 단 하나의 요소도 누락하지 않고 **100% 완전 무결하게 보존**하고 있음이 최종 실증되었다. 인간 디자이너의 임의적인 수작업 축소 및 삭제를 오류로 오인해서는 안 되며, Figma 디자인의 충실도 측면에서 Canvas 엔진의 구현이 올바른 기준(Ground Truth)이다.

  filterFigmaNodeByFrames,
} from '../../../src/pipeline/index';
import {
  simpleDocumentFixture,
  cardLayoutFixture,
  autoLayoutFixture,
} from '../../../src/fixtures/samples';

export function App() {
  const [engine, setEngine] = useState<'canvas' | 'flow'>('canvas');
  const [targetFormat, setTargetFormat] = useState<'docx' | 'hwpx' | 'both'>('both');
  const [mode, setMode] = useState<'balanced' | 'fidelity' | 'editability'>('balanced');
  const [pageOrientation, setPageOrientation] = useState<'portrait' | 'landscape'>('portrait');
  const [inputTab, setInputTab] = useState<'preset' | 'url' | 'docx-to-hwpx'>('preset');
  const [figmaUrl, setFigmaUrl] = useState<string>('');
  const [accessToken, setAccessToken] = useState<string>('');
  const [jsonInput, setJsonInput] = useState<string>('');
  const [selectedFixture, setSelectedFixture] = useState<string>('syn-brief');
  const [status, setStatus] = useState<string>('대기 중');
  const [parseResult, setParseResult] = useState<ParseResult | null>(null);
  const [canvasDoc, setCanvasDoc] = useState<CanvasDocument | null>(null);
  const [docxBlob, setDocxBlob] = useState<Blob | null>(null);
  const [hwpxBlob, setHwpxBlob] = useState<Blob | null>(null);
  const [outputBaseName, setOutputBaseName] = useState<string>('single-page-brief');
  const [duration, setDuration] = useState<number | null>(null);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);
  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [uploadedFileInfo, setUploadedFileInfo] = useState<{
    name: string;
    sizeBytes: number;
    frameCount: number;
  } | null>(null);
  const [loadedSnapshot, setLoadedSnapshot] = useState<{
    targetNode: any;
    imageMap?: Record<string, string>;
    fileName?: string;
    fileKey?: string;
    nodeId?: string;
  } | null>(null);
  const [selectableFrames, setSelectableFrames] = useState<SelectableFrame[]>([]);
  const [selectedFrameIds, setSelectedFrameIds] = useState<string[]>([]);
  const [docxConvertFileInfo, setDocxConvertFileInfo] = useState<{ name: string; size: number } | null>(null);

  const conversionReqIdRef = useRef<number>(0);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const docxFileInputRef = useRef<HTMLInputElement>(null);

  const updateFramesFromNode = (node: any) => {
    try {
      const frames = extractSelectableFrames(node);
      setSelectableFrames(frames);
      setSelectedFrameIds(frames.map((f) => f.id));
    } catch {
      setSelectableFrames([]);
      setSelectedFrameIds([]);
    }
  };

  const handleToggleFrame = (frameId: string) => {
    conversionReqIdRef.current++;
    setSelectedFrameIds((prev) =>
      prev.includes(frameId) ? prev.filter((id) => id !== frameId) : [...prev, frameId]
    );
    setDocxBlob(null);
    setHwpxBlob(null);
  };

  const handleSelectAllFrames = () => {
    conversionReqIdRef.current++;
    setSelectedFrameIds(selectableFrames.map((f) => f.id));
    setDocxBlob(null);
    setHwpxBlob(null);
  };

  const handleDeselectAllFrames = () => {
    conversionReqIdRef.current++;
    setSelectedFrameIds([]);
    setDocxBlob(null);
    setHwpxBlob(null);
  };

  const handleSelectOnlyA4Frames = () => {
    conversionReqIdRef.current++;
    const a4Ids = selectableFrames.filter((f) => f.isRecommendedA4).map((f) => f.id);
    setSelectedFrameIds(a4Ids.length > 0 ? a4Ids : selectableFrames.map((f) => f.id));
    setDocxBlob(null);
    setHwpxBlob(null);
  };

  const handleClearUploadedFile = () => {
    conversionReqIdRef.current++;
    setUploadedFileInfo(null);
    setLoadedSnapshot(null);
    setSelectableFrames([]);
    setSelectedFrameIds([]);
    setDocxBlob(null);
    setHwpxBlob(null);
    setParseResult(null);
    setCanvasDoc(null);
    setStatus('대기 중');
    setErrorMsg(null);
    if (fileInputRef.current) {
      fileInputRef.current.value = '';
    }
  };

  const loadPreset = async (preset: string) => {
    conversionReqIdRef.current++;
    setUploadedFileInfo(null);
    setSelectedFixture(preset);
    setErrorMsg(null);
    let fixtureNode: any = null;
    let baseName = 'document';

    if (preset === 'syn-brief') {
      const data = await import('../../../samples/synthetic-figma/single-page-brief.json');
      fixtureNode = data.default || data;
      baseName = 'single-page-brief';
    } else if (preset === 'syn-report') {
      const data = await import('../../../samples/synthetic-figma/multi-page-report.json');
      fixtureNode = data.default || data;
      baseName = 'multi-page-report';
    } else if (preset === 'syn-portfolio') {
      const data = await import('../../../samples/synthetic-figma/image-centric-portfolio.json');
      fixtureNode = data.default || data;
      baseName = 'image-centric-portfolio';
    } else if (preset === 'syn-article') {
      const data = await import('../../../samples/synthetic-figma/text-centric-article.json');
      fixtureNode = data.default || data;
      baseName = 'text-centric-article';
    } else if (preset === 'syn-infographic') {
      const data = await import('../../../samples/synthetic-figma/vector-centric-infographic.json');
      fixtureNode = data.default || data;
      baseName = 'vector-centric-infographic';
    } else if (preset === 'syn-dashboard') {
      const data = await import('../../../samples/synthetic-figma/complex-nested-dashboard.json');
      fixtureNode = data.default || data;
      baseName = 'complex-nested-dashboard';
    } else if (preset === 'syn-table') {
      const data = await import('../../../samples/synthetic-figma/table-card-hybrid.json');
      fixtureNode = data.default || data;
      baseName = 'table-card-hybrid';
    } else if (preset === 'simple') {
      fixtureNode = simpleDocumentFixture;
      baseName = 'simple-document';
    } else if (preset === 'card') {
      fixtureNode = cardLayoutFixture;
      baseName = 'card-layout';
    } else if (preset === 'auto') {
      fixtureNode = autoLayoutFixture;
      baseName = 'auto-layout';
    } else if (preset === 'e2e-doc') {
      const data = await import('../../../samples/real-world/sample-e2e-document.json');
      fixtureNode = data.default || data;
      baseName = 'sample-e2e-document';
    } else if (preset === 'e2e-card') {
      const data = await import('../../../samples/real-world/sample-e2e-card-grid.json');
      fixtureNode = data.default || data;
      baseName = 'sample-e2e-card-grid';
    } else if (preset === 'e2e-complex') {
      const data = await import('../../../samples/real-world/sample-e2e-complex-layout.json');
      fixtureNode = data.default || data;
      baseName = 'sample-e2e-complex-layout';
    }

    if (fixtureNode) {
      setJsonInput(JSON.stringify(fixtureNode, null, 2));
      setOutputBaseName(baseName);
      const root = 'document' in fixtureNode ? fixtureNode.document : fixtureNode;
      updateFramesFromNode(root);
    }
    setDocxBlob(null);
    setHwpxBlob(null);
  };

  const handleFileUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    conversionReqIdRef.current++;
    setDocxBlob(null);
    setHwpxBlob(null);
    setParseResult(null);
    setCanvasDoc(null);

    const reader = new FileReader();
    reader.onload = (event) => {
      try {
        const text = event.target?.result as string;
        const parsedData = JSON.parse(text);
        let extractedNode: any;
        if (isFigmaSnapshot(parsedData)) {
          extractedNode = parsedData.figmaNode;
          setLoadedSnapshot({
            targetNode: parsedData.figmaNode,
            imageMap: parsedData.imageMap,
            fileName: parsedData.metadata?.fileName || file.name.replace(/\.[^/.]+$/, ''),
            fileKey: parsedData.metadata?.fileKey,
            nodeId: parsedData.metadata?.nodeId,
          });
          updateFramesFromNode(parsedData.figmaNode);
          setStatus(`Figma 스냅샷 [${parsedData.metadata?.fileName || file.name}] 로드 완료`);
        } else {
          const targetNode = parsedData.document || parsedData;
          extractedNode = targetNode;
          const imageMap = parsedData.images || parsedData.meta?.images || {};
          setLoadedSnapshot({
            targetNode,
            imageMap,
            fileName: file.name.replace(/\.[^/.]+$/, ''),
          });
          updateFramesFromNode(targetNode);
          setStatus(`Figma JSON [${file.name}] 로드 완료`);
        }
        let frameCount = 0;
        try {
          frameCount = extractSelectableFrames(extractedNode).length;
        } catch {
          frameCount = 0;
        }
        setUploadedFileInfo({
          name: file.name,
          sizeBytes: file.size,
          frameCount,
        });
        setJsonInput('');
        setOutputBaseName(file.name.replace(/\.[^/.]+$/, ''));
        setErrorMsg(null);
      } catch (err: any) {
        setErrorMsg(`JSON 파일 읽기 실패: ${err.message}`);
      }
    };
    reader.readAsText(file);
  };

  const handleDocxUploadAndConvert = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const currentReqId = ++conversionReqIdRef.current;
    setDocxConvertFileInfo({ name: file.name, size: file.size });
    setDocxBlob(null);
    setHwpxBlob(null);
    setCanvasDoc(null);
    setIsLoading(true);
    setErrorMsg(null);
    setStatus(`DOCX 바이너리 파싱 및 HWPX 변환 중...`);
    const startTime = performance.now();

    try {
      const arrayBuffer = await file.arrayBuffer();
      const docxParser = new DocxCanvasParser();
      const parsedDoc = await docxParser.parse(arrayBuffer);

      if (currentReqId !== conversionReqIdRef.current) return;

      const hwpxRenderer = new HwpxCanvasRenderer();
      const generatedHwpxBlob = await hwpxRenderer.renderToBlob(parsedDoc);

      if (currentReqId !== conversionReqIdRef.current) return;
      const endTime = performance.now();

      setCanvasDoc(parsedDoc);
      setHwpxBlob(generatedHwpxBlob);
      setOutputBaseName(file.name.replace(/\.docx$/i, ''));
      setDuration(Math.round(endTime - startTime));
      setStatus(`DOCX → HWPX 직접 변환 성공 (${parsedDoc.pages.length}페이지, ${parsedDoc.pages.reduce((acc, p) => acc + p.elements.length, 0)}개 요소)`);
    } catch (err: any) {
      if (currentReqId !== conversionReqIdRef.current) return;
      console.error(err);
      setErrorMsg(`DOCX 변환 실패: ${err.message}`);
      setStatus('오류 발생');
    } finally {
      if (currentReqId === conversionReqIdRef.current) {
        setIsLoading(false);
      }
    }
  };

  const executeCanvasConversion = async (targetNode: any, imageMap: Record<string, string>) => {
    const effectiveNode = selectedFrameIds.length > 0 && selectableFrames.length > 0
      ? filterFigmaNodeByFrames(targetNode, selectedFrameIds)
      : targetNode;

    const parser = new FigmaCanvasParser({
      imageMap,
      selectedFrameIds: selectedFrameIds.length > 0 ? selectedFrameIds : undefined,
      titleOverride: outputBaseName,
    });

    const doc = parser.parse(effectiveNode);

    let generatedDocx: Blob | null = null;
    let generatedHwpx: Blob | null = null;

    if (targetFormat === 'docx' || targetFormat === 'both') {
      const docxRenderer = new DocxCanvasRenderer();
      generatedDocx = await docxRenderer.renderToBlob(doc);
    }
    if (targetFormat === 'hwpx' || targetFormat === 'both') {
      const hwpxRenderer = new HwpxCanvasRenderer();
      generatedHwpx = await hwpxRenderer.renderToBlob(doc);
    }

    setCanvasDoc(doc);
    setDocxBlob(generatedDocx);
    setHwpxBlob(generatedHwpx);
  };

  const executeFlowConversion = async (targetNode: any, imageMap: Record<string, string>) => {
    const effectiveNode = selectedFrameIds.length > 0 && selectableFrames.length > 0
      ? filterFigmaNodeByFrames(targetNode, selectedFrameIds)
      : targetNode;

    const parser = new FigmaParser({ imageMap });
    const result = parser.parse(effectiveNode);

    if (result.stats.totalElements === 0) {
      throw new Error(`변환 가능한 Figma 요소를 찾지 못했습니다.`);
    }

    result.document.metadata.conversionMode = mode;
    result.document.pageConfig.orientation = pageOrientation;

    const renderer = new DocxRenderer();
    const blob = await renderer.renderToBlob(result.document);

    setParseResult(result);
    setDocxBlob(blob);
    setHwpxBlob(null);
    setCanvasDoc(null);
  };

  const handleConvert = async () => {
    if (isLoading) return;
    const currentReqId = ++conversionReqIdRef.current;
    setDocxBlob(null);
    setHwpxBlob(null);

    try {
      setIsLoading(true);
      setErrorMsg(null);
      setStatus(`오프라인 ${engine === 'canvas' ? 'Canvas 듀얼 엔진' : 'Flow 엔진'} 변환 중...`);
      const startTime = performance.now();

      let targetNode: any;
      let imageMap: Record<string, string> = {};

      if (loadedSnapshot) {
        targetNode = loadedSnapshot.targetNode;
        imageMap = loadedSnapshot.imageMap || {};
      } else {
        const inputContent = jsonInput.trim() || JSON.stringify(simpleDocumentFixture);
        const parsedData = JSON.parse(inputContent);
        if (isFigmaSnapshot(parsedData)) {
          targetNode = parsedData.figmaNode;
          imageMap = parsedData.imageMap || {};
        } else {
          targetNode = parsedData.document || parsedData;
          imageMap = parsedData.images || parsedData.meta?.images || {};
        }
      }

      if (selectableFrames.length > 0 && selectedFrameIds.length === 0) {
        setErrorMsg('변환할 프레임이 선택되지 않았습니다. 최소 1개 이상의 프레임을 선택해주세요.');
        return;
      }

      if (engine === 'canvas') {
        await executeCanvasConversion(targetNode, imageMap);
      } else {
        await executeFlowConversion(targetNode, imageMap);
      }

      if (currentReqId !== conversionReqIdRef.current) return;
      const endTime = performance.now();
      setDuration(Math.round(endTime - startTime));
      setStatus(`변환 성공 (${engine === 'canvas' ? 'Canvas 듀얼 엔진' : 'Flow 엔진'}, ${targetFormat.toUpperCase()})`);
    } catch (err: any) {
      if (currentReqId !== conversionReqIdRef.current) return;
      console.error(err);
      setErrorMsg(err.message || '변환 중 오류가 발생했습니다.');
      setStatus('오류 발생');
    } finally {
      if (currentReqId === conversionReqIdRef.current) {
        setIsLoading(false);
      }
    }
  };

  const handleFetchAndConvert = async () => {
    if (isLoading) return;
    if (!figmaUrl.trim()) {
      setErrorMsg('Figma URL을 입력해주세요.');
      return;
    }
    if (!accessToken.trim()) {
      setErrorMsg('Figma Access Token (PAT)을 입력해주세요.');
      return;
    }

    const currentReqId = ++conversionReqIdRef.current;
    setIsLoading(true);
    setErrorMsg(null);
    setDocxBlob(null);
    setHwpxBlob(null);

    try {
      setStatus('Figma API에서 파일 구조 조회 중...');
      const urlInfo = parseFigmaUrl(figmaUrl);
      const client = new FigmaClient({ accessToken });

      const fileData = await client.fetchFile(urlInfo.fileKey);
      let targetNode = fileData.document;
      if (urlInfo.nodeId) {
        const findNode = (n: any, id: string): any => {
          if (!n) return undefined;
          if (n.id === id || n.id.replace(/:/g, '-') === id || n.id.replace(/-/g, ':') === id) return n;
          if (n.children) {
            for (const child of n.children) {
              const res = findNode(child, id);
              if (res) return res;
            }
          }
          return undefined;
        };
        const found = findNode(fileData.document, urlInfo.nodeId);
        if (found) targetNode = found;
      }

      let imageMap: Record<string, string> = {};
      try {
        imageMap = await client.fetchImageFills(urlInfo.fileKey);
      } catch {
        // Continue if images fail
      }

      setLoadedSnapshot({
        targetNode,
        imageMap,
        fileName: fileData.name || 'figma-document',
        fileKey: urlInfo.fileKey,
        nodeId: urlInfo.nodeId,
      });
      setOutputBaseName(fileData.name || 'figma-document');
      updateFramesFromNode(targetNode);

      if (engine === 'canvas') {
        await executeCanvasConversion(targetNode, imageMap);
      } else {
        await executeFlowConversion(targetNode, imageMap);
      }

      if (currentReqId !== conversionReqIdRef.current) return;
      setStatus(`Figma 실시간 연동 및 변환 완료!`);
    } catch (err: any) {
      if (currentReqId !== conversionReqIdRef.current) return;
      console.error(err);
      setErrorMsg(err.message || 'Figma 실시간 연동 중 오류 발생');
      setStatus('오류 발생');
    } finally {
      if (currentReqId === conversionReqIdRef.current) {
        setIsLoading(false);
      }
    }
  };

  const handleDownloadDocx = () => {
    if (!docxBlob) return;
    const url = URL.createObjectURL(docxBlob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${outputBaseName}.docx`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

  const handleDownloadHwpx = () => {
    if (!hwpxBlob) return;
    const url = URL.createObjectURL(hwpxBlob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${outputBaseName}.hwpx`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  };

  const handleDownloadBoth = () => {
    handleDownloadDocx();
    setTimeout(() => {
      handleDownloadHwpx();
    }, 200);
  };

  const countElementTypes = (doc: CanvasDocument) => {
    let text = 0, card = 0, image = 0, vector = 0, oval = 0;
    for (const page of doc.pages) {
      for (const el of page.elements) {
        if (el.type === 'text') text++;
        else if (el.type === 'card') card++;
        else if (el.type === 'image') image++;
        else if (el.type === 'vector') vector++;
        else if (el.type === 'oval') oval++;
      }
    }
    return { text, card, image, vector, oval, total: text + card + image + vector + oval };
  };

  return (
    <div style={{ maxWidth: 920, margin: '30px auto', fontFamily: 'system-ui, -apple-system, sans-serif', padding: '0 20px', color: '#1f2937' }}>
      <header style={{ borderBottom: '2px solid #e5e7eb', paddingBottom: 16, marginBottom: 24 }}>
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
          <div>
            <h1 style={{ margin: '0 0 6px 0', fontSize: 26, color: '#111827' }}>Figma2Word & Hancom MVP</h1>
            <p style={{ margin: 0, color: '#4b5563', fontSize: 14 }}>
              Figma 2차원 디자인을 편집 가능한 Microsoft Word(.docx) 및 한컴 한글(.hwpx)로 정밀 변환
            </p>
          </div>
          <span style={{ fontSize: 12, padding: '4px 10px', background: '#dbeafe', color: '#1d4ed8', fontWeight: 700, borderRadius: 20 }}>
            Phase 15 Local MVP
          </span>
        </div>
      </header>

      {/* Engine & Format Controls */}
      <section style={{ backgroundColor: '#f0fdf4', border: '1px solid #bbf7d0', padding: 18, borderRadius: 8, marginBottom: 20 }}>
        <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', flexWrap: 'wrap', gap: 14 }}>
          <div>
            <span style={{ fontSize: 13, fontWeight: 700, color: '#166534', display: 'block', marginBottom: 6 }}>변환 렌더링 엔진:</span>
            <div style={{ display: 'flex', gap: 12 }}>
              <label style={{ display: 'flex', alignItems: 'center', gap: 6, cursor: 'pointer', fontSize: 13 }}>
                <input
                  type="radio"
                  name="engine"
                  value="canvas"
                  checked={engine === 'canvas'}
                  onChange={() => { setEngine('canvas'); setDocxBlob(null); setHwpxBlob(null); }}
                />
                <span style={{ fontWeight: engine === 'canvas' ? 700 : 400, color: engine === 'canvas' ? '#15803d' : '#374151' }}>
                  🎯 Canvas IR Dual Engine (권장: 시각 충실도 0.87+)
                </span>
              </label>
              <label style={{ display: 'flex', alignItems: 'center', gap: 6, cursor: 'pointer', fontSize: 13 }}>
                <input
                  type="radio"
                  name="engine"
                  value="flow"
                  checked={engine === 'flow'}
                  onChange={() => { setEngine('flow'); setDocxBlob(null); setHwpxBlob(null); }}
                />
                <span style={{ fontWeight: engine === 'flow' ? 700 : 400, color: engine === 'flow' ? '#15803d' : '#374151' }}>
                  📄 Flow Engine (레거시: Word 전용)
                </span>
              </label>
            </div>
          </div>

          <div>
            <span style={{ fontSize: 13, fontWeight: 700, color: '#166534', display: 'block', marginBottom: 6 }}>출력 포맷:</span>
            <div style={{ display: 'flex', gap: 10 }}>
              <button
                type="button"
                onClick={() => { setTargetFormat('docx'); setDocxBlob(null); setHwpxBlob(null); }}
                style={{
                  padding: '5px 12px',
                  borderRadius: 6,
                  border: targetFormat === 'docx' ? '2px solid #16a34a' : '1px solid #d1d5db',
                  background: targetFormat === 'docx' ? '#dcfce7' : '#fff',
                  fontWeight: targetFormat === 'docx' ? 700 : 500,
                  fontSize: 12,
                  cursor: 'pointer'
                }}
              >
                DOCX (Word)
              </button>
              <button
                type="button"
                onClick={() => { setTargetFormat('hwpx'); setDocxBlob(null); setHwpxBlob(null); }}
                disabled={engine === 'flow'}
                style={{
                  padding: '5px 12px',
                  borderRadius: 6,
                  border: targetFormat === 'hwpx' ? '2px solid #16a34a' : '1px solid #d1d5db',
                  background: targetFormat === 'hwpx' ? '#dcfce7' : '#fff',
                  fontWeight: targetFormat === 'hwpx' ? 700 : 500,
                  fontSize: 12,
                  cursor: engine === 'flow' ? 'not-allowed' : 'pointer',
                  opacity: engine === 'flow' ? 0.5 : 1
                }}
              >
                HWPX (한글)
              </button>
              <button
                type="button"
                onClick={() => { setTargetFormat('both'); setDocxBlob(null); setHwpxBlob(null); }}
                disabled={engine === 'flow'}
                style={{
                  padding: '5px 12px',
                  borderRadius: 6,
                  border: targetFormat === 'both' ? '2px solid #16a34a' : '1px solid #d1d5db',
                  background: targetFormat === 'both' ? '#dcfce7' : '#fff',
                  fontWeight: targetFormat === 'both' ? 700 : 500,
                  fontSize: 12,
                  cursor: engine === 'flow' ? 'not-allowed' : 'pointer',
                  opacity: engine === 'flow' ? 0.5 : 1
                }}
              >
                듀얼 (DOCX + HWPX)
              </button>
            </div>
          </div>
        </div>
      </section>

      {/* Input Selection Tabs */}
      <section style={{ backgroundColor: '#f9fafb', padding: 20, borderRadius: 8, border: '1px solid #e5e7eb', marginBottom: 20 }}>
        <div style={{ display: 'flex', gap: 8, borderBottom: '1px solid #e5e7eb', paddingBottom: 12, marginBottom: 16 }}>
          <button
            type="button"
            onClick={() => setInputTab('preset')}
            style={{ padding: '8px 16px', borderRadius: 6, border: 'none', background: inputTab === 'preset' ? '#2563eb' : '#e5e7eb', color: inputTab === 'preset' ? '#fff' : '#374151', cursor: 'pointer', fontWeight: 600 }}
          >
            Figma 샘플 프리셋 및 JSON
          </button>
          <button
            type="button"
            onClick={() => setInputTab('docx-to-hwpx')}
            style={{ padding: '8px 16px', borderRadius: 6, border: 'none', background: inputTab === 'docx-to-hwpx' ? '#2563eb' : '#e5e7eb', color: inputTab === 'docx-to-hwpx' ? '#fff' : '#374151', cursor: 'pointer', fontWeight: 600 }}
          >
            🔄 DOCX → HWPX 직접 변환
          </button>
          <button
            type="button"
            onClick={() => setInputTab('url')}
            style={{ padding: '8px 16px', borderRadius: 6, border: 'none', background: inputTab === 'url' ? '#2563eb' : '#e5e7eb', color: inputTab === 'url' ? '#fff' : '#374151', cursor: 'pointer', fontWeight: 600 }}
          >
            Figma URL 라이브 연동
          </button>
        </div>

        {inputTab === 'docx-to-hwpx' ? (
          <div>
            <div style={{ padding: 16, background: '#eff6ff', borderRadius: 6, border: '1px solid #bfdbfe', marginBottom: 16 }}>
              <h4 style={{ margin: '0 0 6px 0', fontSize: 14, color: '#1e40af' }}>Word 문서(.docx)를 한컴 한글(.hwpx)로 원클릭 변환</h4>
              <p style={{ margin: 0, fontSize: 12, color: '#3b82f6' }}>
                Figma2Word에서 생성된 DOCX 또는 VML 도형 기반 Word 문서를 업로드하면, Canvas IR로 파싱하여 표준 KS X 6101 HWPX 문서로 즉시 변환합니다.
              </p>
            </div>
            <div style={{ marginBottom: 16 }}>
              <input
                ref={docxFileInputRef}
                type="file"
                accept=".docx"
                onChange={handleDocxUploadAndConvert}
                style={{ fontSize: 13 }}
              />
            </div>
            {docxConvertFileInfo && (
              <div style={{ fontSize: 12, color: '#4b5563' }}>
                선택된 파일: <strong>{docxConvertFileInfo.name}</strong> ({(docxConvertFileInfo.size / 1024).toFixed(1)} KB)
              </div>
            )}
          </div>
        ) : inputTab === 'url' ? (
          <div>
            <div style={{ marginBottom: 12 }}>
              <label style={{ fontSize: 14, fontWeight: 600, display: 'block', marginBottom: 6 }}>
                Figma URL:
              </label>
              <input
                type="text"
                value={figmaUrl}
                onChange={(e) => setFigmaUrl(e.target.value)}
                placeholder="https://www.figma.com/design/:fileKey/:title?node-id=1-2"
                style={{ width: '100%', boxSizing: 'border-box', padding: '8px 12px', fontSize: 13, border: '1px solid #d1d5db', borderRadius: 4 }}
              />
            </div>
            <div style={{ marginBottom: 16 }}>
              <label style={{ fontSize: 14, fontWeight: 600, display: 'block', marginBottom: 6 }}>
                Figma Personal Access Token (PAT):
              </label>
              <input
                type="password"
                value={accessToken}
                onChange={(e) => setAccessToken(e.target.value)}
                placeholder="figd_... (토큰은 서버로 전송되지 않고 브라우저 로컬에서만 사용됩니다)"
                style={{ width: '100%', boxSizing: 'border-box', padding: '8px 12px', fontSize: 13, border: '1px solid #d1d5db', borderRadius: 4 }}
              />
            </div>
            <button
              type="button"
              onClick={handleFetchAndConvert}
              disabled={isLoading}
              style={{ padding: '10px 20px', borderRadius: 6, border: 'none', background: '#16a34a', color: '#fff', fontSize: 14, fontWeight: 700, cursor: isLoading ? 'not-allowed' : 'pointer' }}
            >
              {isLoading ? '연동 및 변환 중...' : 'Figma 실시간 가져오기 & 변환 실행'}
            </button>
          </div>
        ) : (
          <div>
            <h4 style={{ marginTop: 0, marginBottom: 8, fontSize: 14 }}>검증용 프리셋 선택:</h4>
            <div style={{ display: 'flex', flexWrap: 'wrap', gap: 6, marginBottom: 14 }}>
              <span style={{ fontSize: 11, fontWeight: 700, color: '#2563eb', width: '100%', marginTop: 2 }}>
                🎯 Phase 15 합성 아키타입 (Synthetic Benchmarks):
              </span>
              <button type="button" onClick={() => loadPreset('syn-brief')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-brief' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-brief' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                1. 단일 메모 (Brief)
              </button>
              <button type="button" onClick={() => loadPreset('syn-report')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-report' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-report' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                2. 다중 보고서 (Report 3p)
              </button>
              <button type="button" onClick={() => loadPreset('syn-portfolio')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-portfolio' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-portfolio' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                3. 이미지 갤러리 (Portfolio)
              </button>
              <button type="button" onClick={() => loadPreset('syn-article')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-article' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-article' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                4. 텍스트 아티클 (Article)
              </button>
              <button type="button" onClick={() => loadPreset('syn-infographic')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-infographic' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-infographic' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                5. 벡터 인포그래픽 (Vectors)
              </button>
              <button type="button" onClick={() => loadPreset('syn-dashboard')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-dashboard' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-dashboard' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                6. 중첩 대시보드 (Dashboard)
              </button>
              <button type="button" onClick={() => loadPreset('syn-table')} style={{ padding: '5px 10px', borderRadius: 4, border: '1px solid #2563eb', background: selectedFixture === 'syn-table' ? '#2563eb' : '#eff6ff', color: selectedFixture === 'syn-table' ? '#fff' : '#1d4ed8', cursor: 'pointer', fontSize: 12, fontWeight: 600 }}>
                7. 표/카드 하이브리드 (Table-Card)
              </button>

              <span style={{ fontSize: 11, fontWeight: 700, color: '#4b5563', width: '100%', marginTop: 8 }}>
                📁 E2E 및 기본 샘플:
              </span>
              <button type="button" onClick={() => loadPreset('e2e-doc')} style={{ padding: '4px 8px', borderRadius: 4, border: '1px solid #d1d5db', background: selectedFixture === 'e2e-doc' ? '#4b5563' : '#fff', color: selectedFixture === 'e2e-doc' ? '#fff' : '#374151', cursor: 'pointer', fontSize: 11 }}>
                E2E: 문서형
              </button>
              <button type="button" onClick={() => loadPreset('e2e-card')} style={{ padding: '4px 8px', borderRadius: 4, border: '1px solid #d1d5db', background: selectedFixture === 'e2e-card' ? '#4b5563' : '#fff', color: selectedFixture === 'e2e-card' ? '#fff' : '#374151', cursor: 'pointer', fontSize: 11 }}>
                E2E: 카드형
              </button>
              <button type="button" onClick={() => loadPreset('e2e-complex')} style={{ padding: '4px 8px', borderRadius: 4, border: '1px solid #d1d5db', background: selectedFixture === 'e2e-complex' ? '#4b5563' : '#fff', color: selectedFixture === 'e2e-complex' ? '#fff' : '#374151', cursor: 'pointer', fontSize: 11 }}>
                E2E: 복합형
              </button>
              <button type="button" onClick={() => loadPreset('simple')} style={{ padding: '4px 8px', borderRadius: 4, border: '1px solid #d1d5db', background: selectedFixture === 'simple' ? '#4b5563' : '#fff', color: selectedFixture === 'simple' ? '#fff' : '#374151', cursor: 'pointer', fontSize: 11 }}>
                기본: 심플
              </button>
            </div>

            <div style={{ marginBottom: 12 }}>
              <label style={{ fontSize: 13, fontWeight: 500, display: 'block', marginBottom: 4 }}>
                {uploadedFileInfo ? '다른 Figma JSON 파일로 교체:' : '또는 로컬 Figma JSON 파일 업로드:'}
              </label>
              <input ref={fileInputRef} type="file" accept=".json" onChange={handleFileUpload} />
            </div>

            {uploadedFileInfo ? (
              <div style={{ marginBottom: 16, padding: '10px 14px', background: '#f0fdf4', border: '1px solid #bbf7d0', borderRadius: 6, display: 'flex', justifyContent: 'space-between', alignItems: 'center' }}>
                <span style={{ fontSize: 13, color: '#166534', fontWeight: 600 }}>
                  📁 {uploadedFileInfo.name} ({(uploadedFileInfo.sizeBytes / 1024).toFixed(1)} KB, 프레임 {uploadedFileInfo.frameCount}개)
                </span>
                <button type="button" onClick={handleClearUploadedFile} style={{ padding: '3px 8px', fontSize: 11, borderRadius: 4, border: '1px solid #d1d5db', background: '#fff', cursor: 'pointer' }}>
                  해제
                </button>
              </div>
            ) : (
              <div style={{ marginBottom: 12 }}>
                <textarea
                  value={jsonInput}
                  onChange={(e) => setJsonInput(e.target.value)}
                  placeholder="Figma AST JSON을 여기에 붙여넣으세요..."
                  rows={4}
                  style={{ width: '100%', boxSizing: 'border-box', fontFamily: 'monospace', fontSize: 11, padding: 8, border: '1px solid #d1d5db', borderRadius: 4 }}
                />
              </div>
            )}

            <button
              type="button"
              onClick={handleConvert}
              disabled={isLoading}
              style={{ padding: '8px 20px', borderRadius: 6, border: 'none', background: '#2563eb', color: '#fff', fontSize: 14, fontWeight: 600, cursor: isLoading ? 'not-allowed' : 'pointer' }}
            >
              변환 실행
            </button>
          </div>
        )}
      </section>

      {/* Frame Selection (if frames exist) */}
      {selectableFrames.length > 0 && inputTab !== 'docx-to-hwpx' && (
        <section style={{ backgroundColor: '#f9fafb', padding: 14, borderRadius: 8, border: '1px solid #e5e7eb', marginBottom: 20 }}>
          <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: 10, flexWrap: 'wrap', gap: 8 }}>
            <span style={{ fontSize: 13, fontWeight: 700 }}>
              변환 대상 프레임 ({selectedFrameIds.length} / {selectableFrames.length}개 선택됨)
            </span>
            <div style={{ display: 'flex', gap: 6 }}>
              <button type="button" onClick={handleSelectAllFrames} style={{ padding: '3px 8px', fontSize: 11, borderRadius: 4, border: '1px solid #d1d5db', background: '#fff', cursor: 'pointer' }}>
                전체 선택
              </button>
              <button type="button" onClick={handleDeselectAllFrames} style={{ padding: '3px 8px', fontSize: 11, borderRadius: 4, border: '1px solid #d1d5db', background: '#fff', cursor: 'pointer' }}>
                선택 해제
              </button>
              {selectableFrames.some((f) => f.isA4Ratio) && (
                <button type="button" onClick={handleSelectOnlyA4Frames} style={{ padding: '3px 8px', fontSize: 11, borderRadius: 4, border: '1px solid #2563eb', background: '#eff6ff', color: '#1d4ed8', cursor: 'pointer' }}>
                  A4 비율만 선택
                </button>
              )}
            </div>
          </div>
          <div style={{ display: 'flex', flexDirection: 'column', gap: 4, maxHeight: 160, overflowY: 'auto' }}>
            {selectableFrames.map((frame) => {
              const isChecked = selectedFrameIds.includes(frame.id);
              return (
                <label key={frame.id} style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '6px 10px', borderRadius: 4, border: `1px solid ${isChecked ? '#93c5fd' : '#e5e7eb'}`, background: isChecked ? '#eff6ff' : '#fff', cursor: 'pointer', fontSize: 12 }}>
                  <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
                    <input type="checkbox" checked={isChecked} onChange={() => handleToggleFrame(frame.id)} />
                    <span style={{ fontWeight: 600 }}>{frame.name}</span>
                  </div>
                  <span style={{ fontSize: 11, color: '#6b7280' }}>
                    {Math.round(frame.width)} × {Math.round(frame.height)} {frame.isA4Ratio ? '(A4 권장)' : ''}
                  </span>
                </label>
              );
            })}
          </div>
        </section>
      )}

      {/* Error Alert */}
      {errorMsg && (
        <div style={{ padding: '12px 16px', backgroundColor: '#fee2e2', border: '1px solid #f87171', borderRadius: 6, color: '#991b1b', marginBottom: 20, fontSize: 14 }}>
          <strong>오류:</strong> {errorMsg}
        </div>
      )}

      {/* Results & Download Section */}
      <section style={{ backgroundColor: '#f9fafb', padding: 20, borderRadius: 8, border: '1px solid #e5e7eb' }}>
        <h3 style={{ marginTop: 0, marginBottom: 10, fontSize: 16 }}>진행 상태 및 변환 결과</h3>
        <p style={{ margin: '0 0 14px 0', fontSize: 14 }}>
          <strong>상태:</strong> {status} {duration !== null && `(${duration}ms)`}
        </p>

        {/* Action Buttons */}
        <div style={{ display: 'flex', gap: 10, flexWrap: 'wrap', marginBottom: 20 }}>
          {docxBlob && (
            <button
              type="button"
              onClick={handleDownloadDocx}
              style={{ padding: '10px 18px', borderRadius: 6, border: 'none', background: '#059669', color: '#fff', fontSize: 14, fontWeight: 700, cursor: 'pointer', display: 'flex', alignItems: 'center', gap: 6 }}
            >
              📥 Word 문서 (.docx) 다운로드
            </button>
          )}

          {hwpxBlob && (
            <button
              type="button"
              onClick={handleDownloadHwpx}
              style={{ padding: '10px 18px', borderRadius: 6, border: 'none', background: '#0284c7', color: '#fff', fontSize: 14, fontWeight: 700, cursor: 'pointer', display: 'flex', alignItems: 'center', gap: 6 }}
            >
              📥 한컴 한글 문서 (.hwpx) 다운로드
            </button>
          )}

          {docxBlob && hwpxBlob && (
            <button
              type="button"
              onClick={handleDownloadBoth}
              style={{ padding: '10px 18px', borderRadius: 6, border: 'none', background: '#4f46e5', color: '#fff', fontSize: 14, fontWeight: 700, cursor: 'pointer', display: 'flex', alignItems: 'center', gap: 6 }}
            >
              📦 Word + 한글 동시 다운로드
            </button>
          )}
        </div>

        {/* Canvas Engine Diagnostics & Element Stats */}
        {canvasDoc && (() => {
          const stats = countElementTypes(canvasDoc);
          return (
            <div>
              <h4 style={{ margin: '14px 0 8px 0', fontSize: 14 }}>Canvas IR 정밀 구조 통계</h4>
              <div style={{ display: 'grid', gridTemplateColumns: 'repeat(6, 1fr)', gap: 8, marginBottom: 16 }}>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#6b7280' }}>페이지 수</div>
                  <div style={{ fontSize: 16, fontWeight: 700 }}>{canvasDoc.pages.length}</div>
                </div>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#2563eb' }}>텍스트 (Text)</div>
                  <div style={{ fontSize: 16, fontWeight: 700, color: '#2563eb' }}>{stats.text}</div>
                </div>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#16a34a' }}>카드 (Card)</div>
                  <div style={{ fontSize: 16, fontWeight: 700, color: '#16a34a' }}>{stats.card}</div>
                </div>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#d97706' }}>이미지 (Image)</div>
                  <div style={{ fontSize: 16, fontWeight: 700, color: '#d97706' }}>{stats.image}</div>
                </div>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#7c3aed' }}>벡터 (Vector)</div>
                  <div style={{ fontSize: 16, fontWeight: 700, color: '#7c3aed' }}>{stats.vector}</div>
                </div>
                <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                  <div style={{ fontSize: 10, color: '#059669' }}>원형 (Oval)</div>
                  <div style={{ fontSize: 16, fontWeight: 700, color: '#059669' }}>{stats.oval}</div>
                </div>
              </div>

              {canvasDoc.diagnostics.warnings.length > 0 && (
                <div style={{ marginTop: 10, padding: 8, background: '#fffbeb', border: '1px solid #fef3c7', borderRadius: 4, fontSize: 12 }}>
                  <strong style={{ color: '#92400e' }}>경고 사항:</strong>
                  <ul style={{ margin: '4px 0 0 16px', padding: 0 }}>
                    {canvasDoc.diagnostics.warnings.map((w, idx) => (
                      <li key={idx} style={{ color: '#b45309' }}>{w}</li>
                    ))}
                  </ul>
                </div>
              )}
            </div>
          );
        })()}

        {/* Flow Engine Stats (if flow engine used) */}
        {parseResult && (
          <div>
            <h4 style={{ margin: '14px 0 8px 0', fontSize: 14 }}>Flow 파싱 통계</h4>
            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(4, 1fr)', gap: 8 }}>
              <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                <div style={{ fontSize: 10, color: '#6b7280' }}>전체 노드</div>
                <div style={{ fontSize: 16, fontWeight: 700 }}>{parseResult.stats.totalNodes}</div>
              </div>
              <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                <div style={{ fontSize: 10, color: '#16a34a' }}>지원</div>
                <div style={{ fontSize: 16, fontWeight: 700, color: '#16a34a' }}>{parseResult.stats.supportedNodes}</div>
              </div>
              <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                <div style={{ fontSize: 10, color: '#2563eb' }}>부분 지원</div>
                <div style={{ fontSize: 16, fontWeight: 700, color: '#2563eb' }}>{parseResult.stats.partiallySupportedNodes}</div>
              </div>
              <div style={{ background: '#fff', padding: 8, borderRadius: 4, border: '1px solid #e5e7eb', textAlign: 'center' }}>
                <div style={{ fontSize: 10, color: '#dc2626' }}>미지원</div>
                <div style={{ fontSize: 16, fontWeight: 700, color: '#dc2626' }}>{parseResult.stats.unsupportedNodes}</div>
              </div>
            </div>
          </div>
        )}
      </section>
    </div>
  );
}

export default App;
